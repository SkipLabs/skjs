\begin{abstract}
We formalize the internal type representation used by SKJS and study two related
type transformations:
(1) predicate-driven \emph{refinement} (control-flow narrowing), and
(2) the type-difference operation ($\diff$) used during type inference.
We state and prove soundness properties for these transformations, while
explicitly documenting the intended sources of incompleteness.
\end{abstract}

\section{Runtime Model and Semantic Interface}

\begin{definition}[Runtime Tags]
\label{def:runtime-tags}
Let $\Tags$ be the set of runtime tags (the coarse runtime kind of a value). In the current implementation, tags are represented as integers, so one may take $\Tags = \{0,1,\ldots,11\}$.
\end{definition}

\begin{definition}[Runtime Values and Tags]
\label{def:runtime-values-tags}
Assume runtime values carry their tag as part of the representation: let $\Payload$ be an abstract set of payloads and define
\[
\Val = \Tags \times \Payload.
\]
Define the runtime tag function $\rtag : \Val \to \Tags$ as the first projection:
\[
\rtag((k,p)) = k.
\]
\end{definition}

\begin{definition}[Simple Types]
\label{def:simple-types}
Let $\SType$ be the set of \emph{simple types}, i.e. the atomic type forms that have a single runtime tag.
Formally, $\SType$ is the smallest set generated by the following constructors:
\begin{itemize}
\item primitive atoms: \texttt{undefined}, \texttt{void}, \texttt{null}, \texttt{boolean}, \texttt{number}, \texttt{string}, \texttt{bigint}
\item boolean literals: \texttt{true}, \texttt{false}
\item numeric literals: $n$ for $n \in \mathbb{R}$
\item string literals: \text{``}x\text{''} for strings $x$
\item structured atoms: \texttt{tuple}$(\ldots)$, \texttt{Promise}$(\ldots)$, \texttt{object}$(\ldots)$, \texttt{algebraic}$(\ldots)$ (tagged unions/ADTs with a runtime discriminant), \texttt{function}$(\ldots)$
\end{itemize}
We write these constructors suggestively; most internal parameters are left abstract.
However, for control-flow refinement we will require a concrete semantic interface for object shapes (fields and index signatures); see Definition~\ref{def:object-semantics}.
\end{definition}

\begin{definition}[Simple Types and Tags]
\label{def:simple-types-and-tags}
The function $\tagfn : \SType \to \Tags$ assigns tags:
\begin{align*}
\tagfn(\texttt{undefined}) &= 0 & \tagfn(\texttt{void}) &= 1 \\
\tagfn(\texttt{null}) &= 2 & \tagfn(\texttt{boolean}), \tagfn(\texttt{true}), \tagfn(\texttt{false}) &= 3 \\
\tagfn(\texttt{number}), \tagfn(n) &= 4 \text{ for } n \in \mathbb{R} & 
\tagfn(\texttt{string}), \tagfn(\text{``}x\text{''}) &= 5 \\
\tagfn(\texttt{bigint}) &= 6 & \tagfn(\texttt{tuple}(\ldots)) &= 7 \\
\tagfn(\texttt{Promise}(\ldots)) &= 8 & \tagfn(\texttt{object}(\ldots)) &= 9 \\
\tagfn(\texttt{algebraic}(\ldots)) &= 10 & \tagfn(\texttt{function}(\ldots)) &= 11
\end{align*}
The semantic interpretation $\sem{\cdot} : \SType \to \mathcal{P}(\Val)$ 
maps each simple type to its set of inhabitants.
\paragraph{Note.}
We do \emph{not} define $\sem{\cdot}$ in this document; it is treated as an abstract interpretation function.
Instead, we state explicit assumptions about its behavior as axioms (A0--A1).
\end{definition}

\begin{definition}[\textbf{A0} (Runtime Tag Consistency)]
\label{ax:A0}
The semantic interpretation of simple types agrees with runtime tags:
\[
\forall s \in \SType.\; \forall v \in \Val.\; v \in \sem{s} \implies \rtag(v) = \tagfn(s).
\]
\end{definition}

\begin{definition}[\textbf{A1} (Tag Disjointness)]
\label{ax:A1}
\[
\forall s_1, s_2 \in \SType.\; \tagfn(s_1) \neq \tagfn(s_2) \implies \sem{s_1} \cap \sem{s_2} = \emptyset
\]
\end{definition}

\begin{definition}[Object Values and Object-Type Semantics]
\label{def:object-semantics}
Let $\ObjVal \subseteq \Val$ be the set of object-tagged runtime values:
\[
\ObjVal \triangleq \{ v \in \Val \mid \rtag(v)=9\}.
\]
Assume a (partial) finite-property interface on objects:
\begin{align*}
\mathsf{props} &: \ObjVal \to \mathcal{P}_{\mathrm{fin}}(\String) \\
\mathsf{get} &: \ObjVal \times \String \rightharpoonup \Val
\end{align*}
such that $\mathsf{get}(o,p)$ is defined exactly when $p \in \mathsf{props}(o)$.
Write $\mathsf{has}(o,p)$ for $p \in \mathsf{props}(o)$.

\paragraph{Object field specifications.}
An \emph{object field specification} is a triple $F=(F_{\mathrm{req}},F_{\mathrm{opt}},F_{\mathrm{idx}})$ where:
\begin{itemize}
\item $F_{\mathrm{req}} : \String \rightharpoonup \TType$ is a finite partial map of required fields,
\item $F_{\mathrm{opt}} : \String \rightharpoonup \TType$ is a finite partial map of optional fields,
\item $F_{\mathrm{idx}}$ is either $\mathsf{None}$ or $\mathsf{Some}(K,V)$ for $K,V \in \TType$ (an index signature).
\end{itemize}

\paragraph{Semantics of \texttt{object}$(F)$.}
We assume the object constructor in $\SType$ can carry such an $F$, and define:
\[
o \in \sem{\texttt{object}(F)}
\]
if and only if $o \in \ObjVal$ and the following constraints hold:
\begin{itemize}
\item \textbf{Required fields:} for every $p \in \mathrm{dom}(F_{\mathrm{req}})$,
$\mathsf{has}(o,p)$ and $\mathsf{get}(o,p) \in \sem{F_{\mathrm{req}}(p)}$.
\item \textbf{Optional fields:} for every $p \in \mathrm{dom}(F_{\mathrm{opt}})$,
if $\mathsf{has}(o,p)$ then $\mathsf{get}(o,p) \in \sem{F_{\mathrm{opt}}(p)}$ (and if not, no constraint).
\item \textbf{Index signature (open objects):} if $F_{\mathrm{idx}}=\mathsf{Some}(K,V)$ then for every $p \in \mathsf{props}(o)$:
if the runtime string value $\text{``}p\text{''}$ is in $\sem{K}$, then $\mathsf{get}(o,p) \in \sem{V}$.
\end{itemize}
\paragraph{Note.}
This semantics is intentionally \emph{open}: objects may have properties not mentioned in $F_{\mathrm{req}}$ or $F_{\mathrm{opt}}$.
\end{definition}

\section{Internal Type Representation}

\begin{definition}[STypes]
\label{def:stypes}
$\STypes$ is a finite partial map $\Tags \rightharpoonup \SType$ 
with at most one entry per tag: if $S(k) = s$ then $\tagfn(s) = k$.
\paragraph{Partial-map conventions.}
We write $\mathrm{dom}(S)$ for the domain of defined keys.
When a definition returns ``undefined'' at a key $k$, that means precisely $k \notin \mathrm{dom}(S)$ (i.e. the partial map has no entry at $k$).
\[
\sem{S} = \bigcup_{k \in \mathrm{dom}(S)} \sem{S(k)}
\]
We write $\tags(S) = \mathrm{dom}(S)$.
We write $\{k \mapsto s\}$ for the singleton partial map with domain $\{k\}$.
\paragraph{Intuition.}
An $\STypes$ value represents the \emph{tag-level} part of a union type: it records, for each runtime tag $k$ that may occur, a single representative simple type $s$ of that tag.
The map shape enforces a normalization invariant (``at most one entry per tag''), which matches the implementation: many operations reason by tags and intentionally do not keep multiple alternatives of the same tag.
\paragraph{Example.}
The union \texttt{number | string} is represented as:
\[
S = \{4 \mapsto \texttt{number},\; 5 \mapsto \texttt{string}\}
\]
and has semantics $\sem{S} = \sem{\texttt{number}} \cup \sem{\texttt{string}}$.
In contrast, \texttt{number} and the numeric literal \texttt{1} share the same tag $4$, so an $\STypes$ value cannot simultaneously store both; this is why, for example, a tag-based ``diff'' cannot precisely express \texttt{number \textbackslash{} 1}.
\end{definition}

\begin{definition}[Type Instances]
\label{def:type-instances}
Let $\TInst$ be the set of type instances (type variables or generic instantiations).
\paragraph{Examples.}
A type variable instance is something like $T$ (a unification variable introduced during type checking; this is \emph{not} a source-level type parameter).
A generic instantiation instance is a named type applied to type arguments, e.g. $\texttt{Promise}\langle \texttt{number} \rangle$ or $\texttt{Array}\langle \texttt{string} \rangle$.

Each instance $i \in \TInst$ has a semantic interpretation $\sem{i} \subseteq \Val$.
\end{definition}

\begin{definition}[Type Instance Sets]
\label{def:type-instance-sets}
Let $\TInsts = \mathcal{P}_{\mathrm{fin}}(\TInst)$ be the set of \emph{finite} sets of instances.
For $I \in \TInsts$:
\[
\sem{I} = \bigcup_{i \in I} \sem{i}.
\]
\end{definition}

\begin{definition}[Type Parameters]
\label{def:type-parameters}
Let $\TParam$ be the set of \emph{type parameters} (generic binders), i.e. names introduced by source-level generic declarations such as $\texttt{function f}\langle T\rangle(\ldots)$ or $\texttt{type Box}\langle T\rangle = \ldots$.
They differ from type-variable instances: type parameters are binders whose meaning is fixed by the caller/instantiation, whereas type-variable instances are inference unknowns whose meaning is solved by the type checker.
\end{definition}

\begin{definition}[Clause]
\label{def:clause}
A clause $c = (\vars, \stypes)$ consists of:
\begin{itemize}
\item $\vars \subseteq \TParam$: a finite set of type parameters
\item $\stypes \in \{\mathsf{None}\} \cup \{\mathsf{Some}(S) \mid S \in \STypes\}$
\end{itemize}
Each type parameter $v \in \TParam$ has an interpretation $\sem{v} \subseteq \Val$.
Semantics:
\[
\sem{(\vars, \stypes)} = \left(\bigcap_{v \in \vars} \sem{v}\right) \cap \sem{\stypes}
\]
where $\sem{\mathsf{None}} = \Val$ (top) and $\sem{\mathsf{Some}(S)} = \sem{S}$.
For $\vars = \emptyset$, $\bigcap_{v \in \emptyset} \sem{v} = \Val$.
\[
\text{Well-formedness: } \stypes = \mathsf{None} \implies \vars = \emptyset \quad (\text{i.e., } (\emptyset,\mathsf{None}) \text{ is the unique top clause}).
\]
\end{definition}

\begin{definition}[Clauses]
\label{def:clauses}
$\Clauses$ is a finite map from $\mathcal{P}_{\mathrm{fin}}(\TParam)$ to $\Clause$, 
with at most one clause per key, and such that keys agree with clause variables: if $C(k) = (\vars,\stypes)$ then $\vars = k$.
\[
\sem{C} = \bigcup_{k \in \mathrm{dom}(C)} \sem{C(k)}
\]
\paragraph{Example.}
Let $S_{\mathsf{num}} = \{4 \mapsto \texttt{number}\}$ and $S_{\mathsf{str}} = \{5 \mapsto \texttt{string}\}$.
Let $A,B \in \TParam$ be two type parameters.
Consider the clauses map $C$ with two entries:
\[
C(\{A\}) = (\{A\}, \mathsf{Some}(S_{\mathsf{num}}))
\qquad
C(\{B\}) = (\{B\}, \mathsf{Some}(S_{\mathsf{str}})).
\]
Then
\[
\sem{C} =
\bigl(\sem{A} \cap \sem{\texttt{number}}\bigr)
\cup
\bigl(\sem{B} \cap \sem{\texttt{string}}\bigr).
\]
To illustrate a non-singleton key, consider instead the single-entry map $C'$ defined by:
\[
C'(\{A,B\}) = (\{A,B\}, \mathsf{Some}(S_{\mathsf{num}})).
\]
Then $\sem{C'} = \sem{A} \cap \sem{B} \cap \sem{\texttt{number}}$, i.e. the clause requires \emph{both} parameters.
\end{definition}

\begin{definition}[TType]
\label{def:ttype}
$\TType = (\insts, \clauses)$ where $\insts \in \TInsts$ and $\clauses \in \Clauses$.
\[
\sem{(\insts, \clauses)} = \sem{\insts} \cup \sem{\clauses}
\]
\paragraph{Schematic.}
It is helpful to keep the following picture in mind:
\[
\underbrace{(\insts,\clauses)}_{\TType}
\quad=\quad
\underbrace{\insts}_{\text{``instance'' part}}
\;\;\cup\;\;
\underbrace{\clauses}_{\text{``conditional'' part}}
\]
and later $\diff_{\TType}$ simply applies $\diff$ componentwise to these two parts.
\paragraph{Example.}
Let $I = \{\texttt{Promise}\langle \texttt{number} \rangle\} \in \TInsts$ and let $C$ be the clauses map from the previous example.
Then $t = (I,C)$ denotes the type whose semantics is:
\[
\sem{t} = \sem{\texttt{Promise}\langle \texttt{number} \rangle} \cup \sem{C}.
\]
\end{definition}

\section{Refinement (Predicate-Driven Narrowing)}

\begin{definition}[Refinement]
\label{def:refinement}
Let $t \in \TType$ and let $P \in \Pred$ be a predicate (a set of runtime values).
A \emph{refinement operator} is a function
\[
\refineop : \TType \times \Pred \to \TType.
\]
We call $\refineop$ \emph{branch-sound} if for all $t \in \TType$ and $P \in \Pred$:
\[
\sem{t} \cap P \subseteq \sem{\refineop(t,P)}.
\]
\paragraph{Remark (precision vs soundness).}
Branch-soundness only requires that the refined type \emph{over-approximates} the values that can occur in the branch.
It does not require $\sem{\refineop(t,P)} \subseteq \sem{t}$, so the refinement is allowed to be conservative (lose information) while remaining sound.
\end{definition}

\paragraph{Implementation correspondence.}
In SKJS, control-flow narrowing is implemented by rewriting the internal type for a local variable using a tag-indexed
filter/map on simple types (see the \texttt{FilterType} transformer and \texttt{setLocal} in the implementation).
This motivates defining refinements compositionally by lifting a transformation on $\SType$ values to $\STypes$, $\Clause$, $\Clauses$, and finally $\TType$.

\begin{definition}[Filter/Map Refinement (Implementation Model)]
\label{def:filter-map-refinement}
Let $P \in \Pred$ be a predicate.
Let $f : \SType \to \{\mathsf{true},\mathsf{false}\}$ be a filter and $g : \SType \to \SType$ be a mapper.
We say $(f,g)$ is \emph{sound for $P$} if for every $s \in \SType$ and every $v \in \sem{s}$:
\[
v \in P \implies (f(s) = \mathsf{true} \ \land\  v \in \sem{g(s)}).
\]
\paragraph{Remark (tag preservation).}
In the implementation, the mapper $g$ used for refinement preserves runtime tags:
$\tagfn(g(s))=\tagfn(s)$.
We do not require this for the semantic statement below, but it ensures the lifted $\STypes$ transformation does not merge distinct tags.
\paragraph{Lift to $\STypes$.}
Define:
\[
\refineop_{\STypes}^{f,g}(S) \triangleq \{\, \tagfn(g(S(k))) \mapsto g(S(k)) \mid k \in \mathrm{dom}(S),\ f(S(k))=\mathsf{true}\,\}.
\]
Operationally, this is exactly the ``filter-map'' over the unique representative per tag.
\paragraph{Lift to $\Clause$.}
For a clause $c=(\vars,\stypes)$ define:
\[
\refineop_{\Clause}^{f,g}(c) \triangleq
\begin{cases}
(\vars,\mathsf{None}) & \text{if } \stypes=\mathsf{None} \\
(\vars,\mathsf{Some}(\refineop_{\STypes}^{f,g}(S))) & \text{if } \stypes=\mathsf{Some}(S)
\end{cases}
\]
and interpret the special case $\mathsf{Some}(\emptyset)$ as $\bot$ (dropped), matching the implementation invariant that bottom clauses are not stored.
\paragraph{Lift to $\Clauses$.}
For $C \in \Clauses$, define $\refineop_{\Clauses}^{f,g}(C)$ by applying $\refineop_{\Clause}^{f,g}$ to every clause and dropping $\bot$ results.
\paragraph{Lift to $\TType$.}
For $t=(\insts,\clauses)$ define:
\[
\refineop^{f,g}(t) \triangleq (\insts,\refineop_{\Clauses}^{f,g}(\clauses)).
\]
This matches the current implementation: refinement rewrites the clause part but does not filter the instance part.
\end{definition}

\begin{lemma}[Branch-Soundness of Filter/Map Refinement]
\label{lem:filter-map-branch-sound}
If $(f,g)$ is sound for $P$, then for all $t \in \TType$:
\[
\sem{t} \cap P \subseteq \sem{\refineop^{f,g}(t)}.
\]
\end{lemma}

\begin{proof}
Let $t=(\insts,\clauses)$.
We prove $\sem{t}\cap P \subseteq \sem{\refineop^{f,g}(t)}$ by cases on membership in $\sem{\insts} \cup \sem{\clauses}$.
\begin{itemize}
\item If $v \in \sem{\insts}\cap P$, then $v \in \sem{\insts} \subseteq \sem{\refineop^{f,g}(t)}$ since refinement leaves $\insts$ unchanged.
\item If $v \in \sem{\clauses}\cap P$, then $v \in \sem{(\vars,\stypes)}$ for some clause $(\vars,\stypes)$ in $\clauses$.
If $\stypes=\mathsf{None}$, then $\refineop_{\Clause}^{f,g}(\vars,\mathsf{None})=(\vars,\mathsf{None})$ and the clause remains top, so $v$ is still included.
If $\stypes=\mathsf{Some}(S)$, then $v \in \sem{S(k)}$ for some $k \in \mathrm{dom}(S)$.
By soundness of $(f,g)$ for $P$ and $v\in P$, we have $f(S(k))=\mathsf{true}$ and $v\in \sem{g(S(k))}$.
Hence $v \in \sem{\refineop_{\STypes}^{f,g}(S)}$, so $v$ is contained in the refined clause and therefore in $\sem{\refineop_{\Clauses}^{f,g}(\clauses)}$.
\end{itemize}
Combining cases yields the desired inclusion.
\end{proof}

\subsection{Algorithmic Refinement Judgements (Inference-Rule Style)}

To present refinement in the idiomatic style used in type-system research, we write judgements that update a typing environment.
Let $\Env$ range over environments mapping local variables to $\TType$.
We write $\Env(x)$ for lookup and $\Env[x \mapsto t]$ for pointwise update.

\paragraph{True/false refinements.}
We define two refinement judgements:
\[
\Env \vdash e \Rightarrow \Env'
\qquad\text{and}\qquad
\Env \vdash e \Leftarrow \Env'
\]
to mean ``refine under the assumption that $e$ evaluates to \texttt{true}/\texttt{false}'' respectively.
In this note we only give rules for the refinement forms implemented in SKJS; all other expressions leave the environment unchanged.

\paragraph{Local update helper.}
Given a local $x$ and a refinement transformer $R : \TType \to \TType$, define:
\[
\mathsf{upd}(\Env,x,R) \triangleq \Env[x \mapsto R(\Env(x))].
\]
When a rule refines a non-local expression, it produces $\Env$.

\begin{mathpar}
\inferrule*[right=R-OtherT]{ }{\Env \vdash e \Rightarrow \Env}
\and
\inferrule*[right=R-OtherF]{ }{\Env \vdash e \Leftarrow \Env}
\end{mathpar}

\paragraph{Nullish equality.}
\begin{mathpar}
\inferrule*[right=R-NullT]{ }{\Env \vdash (x === \texttt{null}) \Rightarrow \mathsf{upd}(\Env,x,\,t \mapsto \refineop^{f_{\mathsf{null}},g_{\mathsf{null}}}(t))}
\and
\inferrule*[right=R-NullF]{ }{\Env \vdash (x === \texttt{null}) \Leftarrow \mathsf{upd}(\Env,x,\,t \mapsto \refineop^{f_{\neg\mathsf{null}},g_{\mathsf{null}}}(t))}
\and
\inferrule*[right=R-UndefT]{ }{\Env \vdash (x === \texttt{undefined}) \Rightarrow \mathsf{upd}(\Env,x,\,t \mapsto \refineop^{f_{\mathsf{undef}},g_{\mathsf{undef}}}(t))}
\and
\inferrule*[right=R-UndefF]{ }{\Env \vdash (x === \texttt{undefined}) \Leftarrow \mathsf{upd}(\Env,x,\,t \mapsto \refineop^{f_{\neg\mathsf{undef}},g_{\mathsf{undef}}}(t))}
\end{mathpar}

\paragraph{\texttt{typeof}.}
\begin{mathpar}
\inferrule*[right=R-TypeofT]{ }{\Env \vdash (\texttt{typeof}\ x === k) \Rightarrow \mathsf{upd}(\Env,x,\,t \mapsto \refineop^{f_{P_k},g_{P_k}}(t))}
\and
\inferrule*[right=R-TypeofF]{ }{\Env \vdash (\texttt{typeof}\ x === k) \Leftarrow \mathsf{upd}(\Env,x,\,t \mapsto \refineop^{\neg f_{P_k},g_{P_k}}(t))}
\end{mathpar}
where $P_k$ is the corresponding predicate from Definition~\ref{def:typeof-predicates}.

\paragraph{\texttt{in}.}
\begin{mathpar}
\inferrule*[right=R-InT]{ }{\Env \vdash (\texttt{"}p\texttt{" in }x) \Rightarrow \mathsf{upd}(\Env,x,\,t \mapsto \refineop^{f^{\mathsf{true}}_p,g^{\mathsf{true}}_p}(t))}
\and
\inferrule*[right=R-InF]{ }{\Env \vdash (\texttt{"}p\texttt{" in }x) \Leftarrow \mathsf{upd}(\Env,x,\,t \mapsto \refineop^{f^{\mathsf{false}}_p,g^{\mathsf{false}}_p}(t))}
\end{mathpar}

\paragraph{Boolean connectives.}
The implementation composes refinements sequentially. For example:
\begin{mathpar}
\inferrule*[right=R-AndT]
{\Env \vdash e_1 \Rightarrow \Env_1 \\ \Env_1 \vdash e_2 \Rightarrow \Env_2}
{\Env \vdash (e_1 \ \&\&\ e_2) \Rightarrow \Env_2}
\and
\inferrule*[right=R-OrF]
{\Env \vdash e_1 \Leftarrow \Env_1 \\ \Env_1 \vdash e_2 \Leftarrow \Env_2}
{\Env \vdash (e_1 \ ||\ e_2) \Leftarrow \Env_2}
\end{mathpar}

\paragraph{Soundness connection.}
Each of the case-study lemmas in the following subsections proves that the corresponding rule is branch-sound, via Lemma~\ref{lem:filter-map-branch-sound}.

\subsection{Refinement Case Study: the \texttt{in} Operator}

We formalize the refinement performed for conditions of the form \texttt{"p" in x} in SKJS.
The implementation refines only local variables and updates object types by either (a) adding a required field on the true branch, or (b) forgetting a field on the false branch.

\begin{definition}[\texttt{in} Predicates]
\label{def:in-predicate}
Fix a property name $p \in \String$.
Define:
\[
P_{\mathrm{in}}^{\mathsf{true}}(p) \triangleq \{\, o \in \Val \mid o \in \ObjVal \land \mathsf{has}(o,p)\,\}
\qquad
P_{\mathrm{in}}^{\mathsf{false}}(p) \triangleq \Val \setminus P_{\mathrm{in}}^{\mathsf{true}}(p).
\]
\end{definition}

\begin{definition}[\texttt{in} Filter/Map Pair]
\label{def:in-filter-map}
Fix $p \in \String$ and let $\top_{\TType}$ be any type with $\sem{\top_{\TType}}=\Val$ (e.g. the unique top clause).
Define a mapper $g^{\mathsf{true}}_p : \SType \to \SType$ as follows:
\begin{itemize}
\item If $s$ is an object type $s=\texttt{object}(F)$ and $F_{\mathrm{idx}}=\mathsf{None}$ and $p \notin \mathrm{dom}(F_{\mathrm{req}})$, then $g^{\mathsf{true}}_p(s)$ is $\texttt{object}(F')$ where $F'$ is $F$ except that $F'_{\mathrm{req}}$ additionally maps $p \mapsto \top_{\TType}$.
\item Otherwise, $g^{\mathsf{true}}_p(s)=s$.
\end{itemize}
Let $f^{\mathsf{true}}_p : \SType \to \{\mathsf{true},\mathsf{false}\}$ be the filter that keeps object-tagged structured types and drops primitive atoms and primitive values (matching the implementation's conservative approximation); for the purpose of this note it suffices that:
\[
f^{\mathsf{true}}_p(s)=\mathsf{false} \text{ for primitive/unit/value } s,\quad\text{and}\quad f^{\mathsf{true}}_p(\texttt{object}(F))=\mathsf{true}.
\]

Define a mapper $g^{\mathsf{false}}_p : \SType \to \SType$ by:
\begin{itemize}
\item If $s=\texttt{object}(F)$ then $g^{\mathsf{false}}_p(s)=\texttt{object}(F\setminus p)$ where $F\setminus p$ removes $p$ from both $F_{\mathrm{req}}$ and $F_{\mathrm{opt}}$ (and leaves the index signature unchanged).
\item Otherwise $g^{\mathsf{false}}_p(s)=s$.
\end{itemize}
Let $f^{\mathsf{false}}_p(s)=\mathsf{true}$ for all $s$ (the implementation does not attempt to exclude non-objects in the false branch).
\end{definition}

\begin{lemma}[\texttt{in}-True Branch is Branch-Sound]
\label{lem:in-true-branch-sound}
For every $p \in \String$, the refinement $\refineop^{f^{\mathsf{true}}_p,g^{\mathsf{true}}_p}$ is branch-sound for $P_{\mathrm{in}}^{\mathsf{true}}(p)$.
\end{lemma}

\begin{proof}
We show that $(f^{\mathsf{true}}_p,g^{\mathsf{true}}_p)$ is sound for $P_{\mathrm{in}}^{\mathsf{true}}(p)$ in the sense of Definition~\ref{def:filter-map-refinement}, and conclude by Lemma~\ref{lem:filter-map-branch-sound}.
Let $s \in \SType$ and $v \in \sem{s}$ with $v \in P_{\mathrm{in}}^{\mathsf{true}}(p)$.
Then $v \in \ObjVal$, so $\rtag(v)=9$.
By A0, $\rtag(v)=\tagfn(s)$, hence $\tagfn(s)=9$ and therefore $s$ must be an object type of the form $\texttt{object}(F)$.
By definition of $P_{\mathrm{in}}^{\mathsf{true}}(p)$, $\mathsf{has}(v,p)$.
The filter keeps object types, so $f^{\mathsf{true}}_p(s)=\mathsf{true}$.
If $g^{\mathsf{true}}_p$ adds the required field $p \mapsto \top_{\TType}$, then $v$ satisfies the added requirement because it already has $p$ and $\sem{\top_{\TType}}=\Val$.
Otherwise $g^{\mathsf{true}}_p(s)=s$ and membership is immediate.
Thus $v \in \sem{g^{\mathsf{true}}_p(s)}$ and soundness holds.
\end{proof}

\begin{lemma}[\texttt{in}-False Branch is Branch-Sound]
\label{lem:in-false-branch-sound}
For every $p \in \String$, the refinement $\refineop^{f^{\mathsf{false}}_p,g^{\mathsf{false}}_p}$ is branch-sound for $P_{\mathrm{in}}^{\mathsf{false}}(p)$.
\end{lemma}

\begin{proof}
As above, it suffices to show $(f^{\mathsf{false}}_p,g^{\mathsf{false}}_p)$ is sound for $P_{\mathrm{in}}^{\mathsf{false}}(p)$.
Let $s \in \SType$ and $v \in \sem{s}$ with $v \in P_{\mathrm{in}}^{\mathsf{false}}(p)$.
We have $f^{\mathsf{false}}_p(s)=\mathsf{true}$ by definition.
If $s$ is not an object type, then $g^{\mathsf{false}}_p(s)=s$ so $v \in \sem{g^{\mathsf{false}}_p(s)}$.
If $s=\texttt{object}(F)$, then $g^{\mathsf{false}}_p(s)=\texttt{object}(F\setminus p)$ which removes constraints.
By Definition~\ref{def:object-semantics}, dropping field constraints can only enlarge the set of inhabitants, so $\sem{\texttt{object}(F)} \subseteq \sem{\texttt{object}(F\setminus p)}$ and hence $v \in \sem{g^{\mathsf{false}}_p(s)}$.
\end{proof}

\subsection{Refinement Case Study: Nullish Checks}

We model the refinements performed for comparisons of a local variable against \texttt{null} and \texttt{undefined}.
In the implementation, \texttt{undefined} and \texttt{void} are treated equivalently for refinement.

\begin{definition}[Nullish Predicates]
\label{def:nullish-predicates}
Define the following predicates on runtime values:
\[
P_{\mathsf{null}} \triangleq \{ v \in \Val \mid \rtag(v)=2 \}
\qquad
P_{\mathsf{undef}} \triangleq \{ v \in \Val \mid \rtag(v)\in\{0,1\} \}.
\]
We also write $\neg P$ for $\Val \setminus P$.
\end{definition}

\begin{lemma}[Null Refinement is Branch-Sound]
\label{lem:null-refine-branch-sound}
Let $f_{\mathsf{null}}(s) \triangleq (\tagfn(s)=2)$ and let $g_{\mathsf{null}}(s)\triangleq s$.
Then $\refineop^{f_{\mathsf{null}},g_{\mathsf{null}}}$ is branch-sound for $P_{\mathsf{null}}$.
Moreover, letting $f_{\neg\mathsf{null}}(s)\triangleq (\tagfn(s)\neq 2)$, $\refineop^{f_{\neg\mathsf{null}},g_{\mathsf{null}}}$ is branch-sound for $\neg P_{\mathsf{null}}$.
\end{lemma}

\begin{proof}
We prove soundness of $(f_{\mathsf{null}},g_{\mathsf{null}})$ for $P_{\mathsf{null}}$ and apply Lemma~\ref{lem:filter-map-branch-sound}.
Let $s \in \SType$ and $v \in \sem{s}$ with $v \in P_{\mathsf{null}}$.
Then $\rtag(v)=2$, and by A0 we have $\tagfn(s)=2$, hence $f_{\mathsf{null}}(s)=\mathsf{true}$.
Since $g_{\mathsf{null}}$ is the identity, $v \in \sem{g_{\mathsf{null}}(s)}$ is immediate.
The $\neg P_{\mathsf{null}}$ case is analogous.
\end{proof}

\begin{lemma}[Undefined/Void Refinement is Branch-Sound]
\label{lem:undef-refine-branch-sound}
Let $f_{\mathsf{undef}}(s) \triangleq (\tagfn(s)\in\{0,1\})$ and $g_{\mathsf{undef}}(s)\triangleq s$.
Then $\refineop^{f_{\mathsf{undef}},g_{\mathsf{undef}}}$ is branch-sound for $P_{\mathsf{undef}}$.
Moreover, letting $f_{\neg\mathsf{undef}}(s)\triangleq (\tagfn(s)\notin\{0,1\})$, $\refineop^{f_{\neg\mathsf{undef}},g_{\mathsf{undef}}}$ is branch-sound for $\neg P_{\mathsf{undef}}$.
\end{lemma}

\begin{proof}
As above, we show $(f_{\mathsf{undef}},g_{\mathsf{undef}})$ is sound for $P_{\mathsf{undef}}$.
Let $s \in \SType$ and $v \in \sem{s}$ with $\rtag(v)\in\{0,1\}$.
By A0, $\tagfn(s)=\rtag(v)\in\{0,1\}$, so $f_{\mathsf{undef}}(s)=\mathsf{true}$ and $v\in\sem{g_{\mathsf{undef}}(s)}$.
The $\neg P_{\mathsf{undef}}$ case is analogous.
\end{proof}

\subsection{Refinement Case Study: \texttt{typeof} Checks}

We model the refinements performed for tests of the form \texttt{typeof x === "k"} for a subset of strings $k$.
The implementation refines based on the internal tag/classification of the current $\SType$ alternatives.
\paragraph{Note.}
The predicate modeled here is the one used by the checker for narrowing; it may be a conservative approximation of JavaScript runtime \texttt{typeof}.

\begin{definition}[\texttt{typeof} Predicates (Tag-Level)]
\label{def:typeof-predicates}
Define the following tag-level predicates:
\begin{align*}
P_{\mathsf{typeof}=\texttt{"string"}} &\triangleq \{ v \in \Val \mid \rtag(v)=5 \} \\
P_{\mathsf{typeof}=\texttt{"number"}} &\triangleq \{ v \in \Val \mid \rtag(v)=4 \} \\
P_{\mathsf{typeof}=\texttt{"bigint"}} &\triangleq \{ v \in \Val \mid \rtag(v)=6 \} \\
P_{\mathsf{typeof}=\texttt{"boolean"}} &\triangleq \{ v \in \Val \mid \rtag(v)=3 \} \\
P_{\mathsf{typeof}=\texttt{"function"}} &\triangleq \{ v \in \Val \mid \rtag(v)=11 \} \\
P_{\mathsf{typeof}=\texttt{"undefined"}} &\triangleq \{ v \in \Val \mid \rtag(v)\in\{0,1\} \}.
\end{align*}
\end{definition}

\begin{lemma}[\texttt{typeof} Refinement is Branch-Sound (Tag-Level)]
\label{lem:typeof-refine-branch-sound}
For each predicate $P$ in Definition~\ref{def:typeof-predicates}, define
$f_P(s)\triangleq (\tagfn(s)\in \tags_P)$ where $\tags_P$ is the corresponding set of tags (e.g. $\{5\}$ for \texttt{"string"}, $\{0,1\}$ for \texttt{"undefined"}), and let $g_P(s)\triangleq s$.
Then $\refineop^{f_P,g_P}$ is branch-sound for $P$, and $\refineop^{\neg f_P,g_P}$ is branch-sound for $\neg P$.
\end{lemma}

\begin{proof}
Fix $P$ and let $s \in \SType$ and $v \in \sem{s}$ with $v \in P$.
By A0, $\tagfn(s)=\rtag(v)\in\tags_P$, so $f_P(s)=\mathsf{true}$ and $v\in\sem{g_P(s)}$.
Apply Lemma~\ref{lem:filter-map-branch-sound}.
The $\neg P$ case is analogous.
\end{proof}

\section{Type Difference (Diff)}

\begin{definition}[Diff Soundness Properties]
\label{def:diff-soundness}
Let $X$ be a set equipped with a semantic interpretation $\sem{\cdot} : X \to \mathcal{P}(\Val)$.
An operation $\diff : X \times X \to X$ is \emph{sound} (with respect to $\sem{\cdot}$) if for all $x_1,x_2 \in X$, letting $x_3 = \diff(x_1,x_2)$, the following properties hold:
\begin{align}
\text{(P1)} \quad & \sem{x_3} \subseteq \sem{x_1} \tag{under-approximation} \\
\text{(P2)} \quad & \sem{x_3} \cup \sem{x_2} \supseteq \sem{x_1} \tag{coverage}
\end{align}
In this document we instantiate $X$ with $\STypes$, $\Clause$, $\Clauses$, and $\TType$.
\paragraph{Reader's guide.}
These two properties imply that $\sem{x_3}$ is always a refinement of $\sem{x_1}$, and that it still covers all of $\sem{x_1}$ once $\sem{x_2}$ is added back.
Equivalently (Lemma~\ref{lem:diff-equiv-char}), $\sem{x_3}$ is an \emph{over-approximation} of the true set difference $\sem{x_1}\setminus\sem{x_2}$ while staying within $\sem{x_1}$.
\end{definition}

\begin{lemma}[Equivalent Characterization]
\label{lem:diff-equiv-char}
For all $x_1,x_2,x_3 \in X$, (P1) $\land$ (P2) holds if and only if
\[
\sem{x_1} \setminus \sem{x_2} \subseteq \sem{x_3} \subseteq \sem{x_1}.
\]
\end{lemma}

\begin{proof}
($\Rightarrow$) Let $v \in \sem{x_1} \setminus \sem{x_2}$. By (P2),
$v \in \sem{x_3} \cup \sem{x_2}$. Since $v \notin \sem{x_2}$, we have $v \in \sem{x_3}$.
Thus $\sem{x_1} \setminus \sem{x_2} \subseteq \sem{x_3}$. Combined with (P1):
$\sem{x_1} \setminus \sem{x_2} \subseteq \sem{x_3} \subseteq \sem{x_1}$.

($\Leftarrow$) The right-hand inclusion $\sem{x_3} \subseteq \sem{x_1}$ is exactly (P1).
For (P2),
\[
\sem{x_3} \cup \sem{x_2} \supseteq (\sem{x_1} \setminus \sem{x_2}) \cup \sem{x_2} \supseteq \sem{x_1}.
\]
\end{proof}

\begin{definition}[Completeness]
A diff operation is \emph{complete} if:
\[
\sem{\diff(t_1, t_2)} = \sem{t_1} \setminus \sem{t_2}
\]
By the equivalent characterization, (P1) $\land$ (P2) implies 
$\sem{t_1} \setminus \sem{t_2} \subseteq \sem{\diff(t_1, t_2)}$.
Completeness additionally requires the reverse: $\sem{\diff(t_1, t_2)} \subseteq \sem{t_1} \setminus \sem{t_2}$.
\end{definition}

\subsection{STypes Diff}

\begin{definition}[STypes Diff]
\label{def:stypes-diff}
\[
\diff_{\STypes}(L, R)(k) =
\begin{cases}
L(k) & \text{if } k \in \tags(L) \setminus \tags(R) \\
\text{undefined} & \text{otherwise}
\end{cases}
\]
\paragraph{Intuition.}
Since an $\STypes$ value stores at most one representative per runtime tag, this diff only removes whole tags:
it preserves exactly the tags present on the left but absent on the right.
\end{definition}

\begin{lemma}[STypes Diff Satisfies (P1)]
\label{lem:stypes-diff-p1}
$\sem{\diff_{\STypes}(L, R)} \subseteq \sem{L}$.
\end{lemma}

\begin{proof}
By definition, $\tags(\diff_{\STypes}(L,R)) = \tags(L) \setminus \tags(R) \subseteq \tags(L)$, and for all $k \in \tags(\diff_{\STypes}(L,R))$ we have $\diff_{\STypes}(L,R)(k)=L(k)$.
Therefore:
\[
\sem{\diff_{\STypes}(L,R)}
= \bigcup_{k \in \tags(\diff_{\STypes}(L,R))} \sem{L(k)}
\subseteq \bigcup_{k \in \tags(L)} \sem{L(k)}
= \sem{L}.
\]
\end{proof}

\begin{definition}[Same-Tag Coverage Condition]
\label{def:stc}
For $L, R \in \STypes$, define:
\[
\mathsf{STC}(L, R) \iff \forall k \in \tags(L) \cap \tags(R).\; \sem{L(k)} \subseteq \sem{R(k)}
\]
\end{definition}

\begin{lemma}[STypes Diff Satisfies (P2) under STC]
\label{lem:stypes-diff-p2-under-stc}
If $\mathsf{STC}(L, R)$ holds, then $\sem{\diff_{\STypes}(L, R)} \cup \sem{R} \supseteq \sem{L}$.
\end{lemma}

\begin{proof}
Let $x \in \sem{L}$. Then $x \in \sem{L(k)}$ for some $k \in \tags(L)$.
\begin{itemize}
\item Case $k \notin \tags(R)$: Then $k \in \tags(\diff_{\STypes}(L,R))$ and $\diff_{\STypes}(L,R)(k)=L(k)$, so $x \in \sem{\diff_{\STypes}(L,R)}$.
\item Case $k \in \tags(R)$: By $\mathsf{STC}$, $\sem{L(k)} \subseteq \sem{R(k)} \subseteq \sem{R}$, so $x \in \sem{R}$.
\end{itemize}
\end{proof}

\begin{corollary}[STypes Diff under STC]
\label{cor:stypes-diff-under-stc}
If $\mathsf{STC}(L, R)$ holds, then (P2) holds.
In particular, if $L(k) = R(k)$ for all $k \in \tags(L) \cap \tags(R)$, then (P2) holds.
\end{corollary}

\begin{lemma}[STypes Diff Completeness]
\label{lem:stypes-diff-complete-iff-stc}
$\diff_{\STypes}$ is complete (i.e., $\sem{\diff_{\STypes}(L, R)} = \sem{L} \setminus \sem{R}$) 
if and only if $\mathsf{STC}(L, R)$.
\end{lemma}

\begin{proof}
($\Rightarrow$) Suppose completeness holds and let $k \in \tags(L) \cap \tags(R)$.
Since $k \notin \tags(\diff_{\STypes}(L,R))$, the component $L(k)$ contributes no value to $\sem{L} \setminus \sem{R}$ under completeness, hence $\sem{L(k)} \subseteq \sem{R}$.
By A1 and the ``at most one entry per tag'' invariant for $\STypes$, $\sem{L(k)}$ can only overlap with $\sem{R(k)}$, so $\sem{L(k)} \subseteq \sem{R(k)}$.

($\Leftarrow$) We show $\sem{\diff_{\STypes}(L, R)} = \sem{L} \setminus \sem{R}$.

($\subseteq$) Let $x \in \sem{\diff_{\STypes}(L, R)}$. Then $x \in \sem{L(k)}$ for some $k \in \tags(L) \setminus \tags(R)$.
By A1, $x \notin \sem{R}$ (no type in $R$ has tag $k$), so $x \in \sem{L} \setminus \sem{R}$.

($\supseteq$) Let $x \in \sem{L} \setminus \sem{R}$. Then $x \in \sem{L(k)}$ for some $k \in \tags(L)$ and $x \notin \sem{R}$.
If $k \in \tags(R)$, then by hypothesis $\sem{L(k)} \subseteq \sem{R(k)} \subseteq \sem{R}$, contradicting $x \notin \sem{R}$.
So $k \notin \tags(R)$, meaning $k \in \tags(\diff_{\STypes}(L,R))$ and $x \in \sem{\diff_{\STypes}(L, R)}$.
\end{proof}

\begin{corollary}[When STypes Diff is Incomplete]
$\diff_{\STypes}(L, R)$ is incomplete when there exists $k \in \tags(L) \cap \tags(R)$ with $\sem{L(k)} \not\subseteq \sem{R(k)}$.
Example: $\diff_{\STypes}(\{4 \mapsto \texttt{number}\}, \{4 \mapsto \texttt{1}\}) = \emptyset$, 
but $\sem{\texttt{number}} \setminus \sem{\texttt{1}} \neq \emptyset$.
\end{corollary}

\subsection{TInsts Diff}

\begin{definition}[TInsts Diff]
\label{def:tinsts-diff}
\[
\diff_{\TInsts}(L, R) = L \setminus R = \{ i \in L \mid i \notin R \}
\]
\paragraph{Note.}
In the implementation, $\TInsts$ is represented as a collection of instance AST nodes and $\diff_{\TInsts}$ removes elements based on \emph{structural equality} of instances.
In this paper, that corresponds exactly to the usual mathematical set difference on the set $\TInst$ (i.e. membership/equality is the syntactic identity of instances, not semantic overlap).
\end{definition}

\begin{lemma}[TInsts Diff Satisfies (P1) and (P2)]
\label{lem:tinsts-diff-sound}
$\sem{\diff_{\TInsts}(L, R)} \subseteq \sem{L}$ and 
$\sem{\diff_{\TInsts}(L, R)} \cup \sem{R} \supseteq \sem{L}$.
\end{lemma}

\begin{proof}
(P1): $\diff_{\TInsts}(L, R) \subseteq L$, so $\sem{\diff_{\TInsts}(L, R)} \subseteq \sem{L}$.

(P2): Let $x \in \sem{L}$. Then $x \in \sem{i}$ for some $i \in L$.
\begin{itemize}
\item If $i \in R$: Then $x \in \sem{i} \subseteq \sem{R}$.
\item If $i \notin R$: Then $i \in \diff_{\TInsts}(L, R)$, so $x \in \sem{\diff_{\TInsts}(L, R)}$.
\end{itemize}
\end{proof}

\begin{lemma}[TInsts Diff Completeness]
\label{lem:tinsts-diff-complete}
$\diff_{\TInsts}$ is complete if and only if for all $i \in L \setminus R$ and $j \in R$:
$\sem{i} \cap \sem{j} = \emptyset$.
\end{lemma}

\begin{proof}
($\Rightarrow$) Suppose completeness and let $i \in L \setminus R$ and $j \in R$.
If $x \in \sem{i} \cap \sem{j}$, then $i \in \diff_{\TInsts}(L, R)$, so $x \in \sem{\diff_{\TInsts}(L, R)} = \sem{L} \setminus \sem{R}$.
But $x \in \sem{j} \subseteq \sem{R}$, contradiction.

($\Leftarrow$) We show $\sem{\diff_{\TInsts}(L, R)} = \sem{L} \setminus \sem{R}$.

($\subseteq$) Let $x \in \sem{\diff_{\TInsts}(L, R)}$. Then $x \in \sem{i}$ for some $i \in L \setminus R$.
For any $j \in R$, $\sem{i} \cap \sem{j} = \emptyset$ by hypothesis.
Thus $x \notin \sem{R}$, so $x \in \sem{L} \setminus \sem{R}$.

($\supseteq$) Let $x \in \sem{L} \setminus \sem{R}$. Then $x \in \sem{i}$ for some $i \in L$.
If $i \in R$, then $x \in \sem{R}$, contradiction. So $i \in L \setminus R = \diff_{\TInsts}(L, R)$.
\end{proof}

\subsection{Clause Diff}

\begin{definition}[Clause Diff]
\label{def:clause-diff}
For clauses $c_1 = (\vars_1, \stypes_1)$ and $c_2 = (\vars_2, \stypes_2)$:
\[
\diff_{\Clause}(c_1, c_2) = 
\begin{cases}
\bot & \text{if } \stypes_2 = \mathsf{None} \\
c_1 & \text{if } \stypes_1 = \mathsf{None} \\
(\vars_1, \mathsf{Some}(\diff_{\STypes}(S_1, S_2))) & \text{if } \stypes_1 = \mathsf{Some}(S_1), \stypes_2 = \mathsf{Some}(S_2)
\end{cases}
\]
where $\bot$ represents the empty type (dropped from results) with $\sem{\bot} = \emptyset$.
\end{definition}

\begin{lemma}[Clause Diff Satisfies (P1)]
\label{lem:clause-diff-p1}
$\sem{\diff_{\Clause}(c_1, c_2)} \subseteq \sem{c_1}$.
\end{lemma}

\begin{proof}
Case analysis:
\begin{itemize}
\item $\stypes_2 = \mathsf{None}$: Result is $\bot$. $\sem{\bot} = \emptyset \subseteq \sem{c_1}$.
\item $\stypes_1 = \mathsf{None}$: Result is $c_1$. $\sem{c_1} \subseteq \sem{c_1}$.
\item Both $\mathsf{Some}$: 
$\sem{(\vars_1, \mathsf{Some}(\diff_{\STypes}(S_1, S_2)))} = 
\left(\bigcap_{v \in \vars_1} \sem{v}\right) \cap \sem{\diff_{\STypes}(S_1, S_2)}
\subseteq \left(\bigcap_{v \in \vars_1} \sem{v}\right) \cap \sem{S_1} = \sem{c_1}$.
\end{itemize}
\end{proof}

\begin{lemma}[Clause Diff Satisfies (P2) Conditionally]
\label{lem:clause-diff-p2-conditional}
$\diff_{\Clause}$ satisfies (P2) when:
\begin{itemize}
\item Case $\stypes_2 = \mathsf{None}$: Always for well-formed clauses (since then $c_2$ is the unique top clause and $\sem{c_2} = \Val \supseteq \sem{c_1}$).
\item Case $\stypes_1 = \mathsf{None}$: Always (since result is $c_1$).
\item Case both $\mathsf{Some}$: When $\diff_{\STypes}(S_1, S_2)$ satisfies (P2).
\end{itemize}
\end{lemma}

\subsection{Clauses Diff}

\begin{definition}[Clauses Diff]
\label{def:clauses-diff}
For $C_1, C_2 \in \Clauses$, let $c_{\emptyset} = C_2(\emptyset)$ if $\emptyset \in \mathrm{dom}(C_2)$, else undefined.
\[
\diff_{\Clauses}(C_1, C_2)(k) = 
\begin{cases}
\diff_{\Clause}(\diff_{\Clause}(C_1(k), C_2(k)), c_{\emptyset}) & \text{if } k \in \mathrm{dom}(C_2) \land c_{\emptyset} \text{ defined} \\
\diff_{\Clause}(C_1(k), C_2(k)) & \text{if } k \in \mathrm{dom}(C_2) \land c_{\emptyset} \text{ undefined} \\
\diff_{\Clause}(C_1(k), c_{\emptyset}) & \text{if } k \notin \mathrm{dom}(C_2) \land c_{\emptyset} \text{ defined} \\
C_1(k) & \text{if } k \notin \mathrm{dom}(C_2) \land c_{\emptyset} \text{ undefined}
\end{cases}
\]
for $k \in \mathrm{dom}(C_1)$. Clauses mapped to $\bot$ are dropped.
\paragraph{Note.}
Whenever $k \in \mathrm{dom}(C_2)$, the call $\diff_{\Clause}(C_1(k), C_2(k))$ compares clauses stored under the \emph{same key} $k$, so their variable-sets match by the $\Clauses$ well-formedness condition (keys agree with clause variables).
The only subtraction that may use a different key is via $c_{\emptyset}=C_2(\emptyset)$, which is deliberately applied to every $C_1(k)$; this corresponds to the implementation behavior and explains why the right-hand clause's variables are not used by $\diff_{\Clause}$.
\paragraph{Why the $\emptyset$-clause matters.}
The distinguished key $\emptyset$ corresponds to clauses that do not require any type parameters.
When present on the right, subtracting $c_{\emptyset}$ from every left clause models a ``global'' elimination that should apply regardless of which type-parameter constraints produced the left clause.
\end{definition}

\begin{lemma}[Clauses Diff Satisfies (P1)]
\label{lem:clauses-diff-p1}
$\sem{\diff_{\Clauses}(C_1, C_2)} \subseteq \sem{C_1}$.
\end{lemma}

\begin{proof}
Each output clause is derived from $C_1(k)$ via $\diff_{\Clause}$, which satisfies (P1).
\end{proof}

\subsection{TType Diff}

\begin{definition}[TType Diff]
\label{def:ttype-diff}
For $t_1 = (\insts_1, \clauses_1)$ and $t_2 = (\insts_2, \clauses_2)$:
\[
\diff_{\TType}(t_1, t_2) = (\diff_{\TInsts}(\insts_1, \insts_2), \diff_{\Clauses}(\clauses_1, \clauses_2))
\]
\end{definition}

\begin{theorem}[TType Diff Satisfies (P1)]
\label{thm:ttype-diff-p1}
$\sem{\diff_{\TType}(t_1, t_2)} \subseteq \sem{t_1}$.
\end{theorem}

\begin{proof}
\begin{align*}
\sem{\diff_{\TType}(t_1, t_2)} &= \sem{\diff_{\TInsts}(\insts_1, \insts_2)} \cup \sem{\diff_{\Clauses}(\clauses_1, \clauses_2)} \\
&\subseteq \sem{\insts_1} \cup \sem{\clauses_1} = \sem{t_1}
\end{align*}
by Lemmas~\ref{lem:tinsts-diff-sound} and~\ref{lem:clauses-diff-p1}.
\end{proof}

\begin{theorem}[TType Diff Satisfies (P2) Conditionally]
$\diff_{\TType}$ satisfies (P2) when all underlying $\diff_{\STypes}$ calls satisfy (P2).
In particular, (P2) holds when $\mathsf{STC}$ holds for all same-tag eliminations.
\end{theorem}

\begin{theorem}[TType Diff Completeness]
$\diff_{\TType}$ is complete when:
\begin{enumerate}
\item $\diff_{\TInsts}$ is complete (structurally distinct instances are semantically disjoint), and
\item $\diff_{\STypes}$ is complete for all clause pairs (same-tag implies $\sem{s} \subseteq \sem{s'}$).
\end{enumerate}
\end{theorem}

\begin{corollary}[Explicit Completeness Conditions for $\diff_{\TType}$]
Let $t_1=(\insts_1,\clauses_1)$ and $t_2=(\insts_2,\clauses_2)$.
Then $\diff_{\TType}(t_1,t_2)$ is complete (i.e. $\sem{\diff_{\TType}(t_1,t_2)}=\sem{t_1}\setminus\sem{t_2}$) under the following explicit hypotheses:
\begin{enumerate}
\item \textbf{Instance disjointness across sides:} for all $i\in\insts_1$ and $j\in\insts_2$ with $i\neq j$, we have $\sem{i}\cap\sem{j}=\emptyset$.
\item \textbf{Top-clause alignment:} if $\clauses_1$ is top (i.e. it contains the unique top clause $(\emptyset,\mathsf{None})$), then $\clauses_2$ is also top.
\item \textbf{Same-tag coverage for every clause subtraction:}
  \begin{enumerate}
  \item For every $k \in \mathrm{dom}(\clauses_1)\cap \mathrm{dom}(\clauses_2)$, if $\clauses_1(k)=(k,\mathsf{Some}(S_1))$ and $\clauses_2(k)=(k,\mathsf{Some}(S_2))$, then $\mathsf{STC}(S_1,S_2)$.
  \item If $\emptyset\in \mathrm{dom}(\clauses_2)$ and $\clauses_2(\emptyset)=(\emptyset,\mathsf{Some}(S_0))$, then for every $k\in \mathrm{dom}(\clauses_1)$ with $\clauses_1(k)=(k,\mathsf{Some}(S_1))$, we have $\mathsf{STC}(S_1,S_0)$.
  \end{enumerate}
\end{enumerate}
\end{corollary}

\section{Application to Type Inference}

\begin{definition}[Subtyping Check with Validation]
When checking $u_1 \leq u_2$ where $u_2$ contains type variable $T$:
\begin{enumerate}
\item Compute $\diff(u_1, u_2)$ and grow $T$'s lower bound.
\item Expand $u_1, u_2$ using the substitution.
\item Validate via $\mathsf{isSub}(\mathit{expand}(u_1), \mathit{expand}(u_2))$.
\end{enumerate}
\end{definition}

\begin{theorem}[Inference Soundness]
If $\mathsf{isSub}$ correctly validates subtyping on expanded types, then 
type inference is sound regardless of whether $\diff$ satisfies (P2).
\end{theorem}

\begin{proof}
The $\diff$ result is used only to guide inference (e.g. by accumulating bounds or hints about $T$).
The acceptance/rejection decision is made solely by $\mathsf{isSub}$ on expanded types.
Therefore, an imprecise $\diff$ can at worst cause incompleteness (e.g. by failing to discover a useful bound, leading to a later rejection), but it cannot cause an unsound acceptance when $\mathsf{isSub}$ is correct.
\end{proof}

\section{Examples of Incompleteness}

We present two concrete examples where $\diff$ is incomplete, i.e., 
$\sem{\diff(t_1, t_2)} \neq \sem{t_1} \setminus \sem{t_2}$ (it may be a strict subset or a strict superset of the true set difference).

\subsection{Example 1: Same-Tag Literal Types}

\textbf{TypeScript code:}
\begin{verbatim}
function f(x: 1 | 2): 2 {
  if (x === 1) { return 2; }
  return x;  // x should be narrowed to 2
}
\end{verbatim}

\textbf{Mathematical setup:}
Both \texttt{1} and \texttt{2} have the same runtime tag $4$.
Because $\STypes$ stores at most one $\SType$ per tag, it cannot precisely represent unions of same-tag literals such as \texttt{1 | 2}.
In practice, the implementation merges same-tag literals into a single representative $\SType$ for the tag (often a wider type such as \texttt{number}), which loses the ability to subtract a single literal while keeping another.

\textbf{Incompleteness:}
Since $\tagfn(\texttt{1}) = \tagfn(\texttt{2}) = 4$ and $\STypes$ stores at most one entry per tag,
$\diff_{\STypes}$ cannot distinguish between literals of the same tag.
For a simpler illustration:
\[
\diff_{\STypes}(\{4 \mapsto \texttt{number}\}, \{4 \mapsto \texttt{1}\}) = \emptyset
\]
But $\sem{\texttt{number}} \setminus \sem{\texttt{1}} = \text{all numbers except 1} \neq \emptyset$.

The condition $\mathsf{STC}(L, R)$ fails: $\sem{\texttt{number}} \not\subseteq \sem{\texttt{1}}$.

\subsection{Example 2: Top Type (unknown)}

\textbf{TypeScript code (requires \texttt{--strictNullChecks}):}
\begin{verbatim}
function f(x: unknown): {} {
  if (x !== null && x !== undefined) {
    return x;  // x should be narrowed to {}
  }
  return {};
}
\end{verbatim}

\textbf{Mathematical setup:}
The type \texttt{unknown} is represented as a clause with $\stypes = \mathsf{None}$ (top).
Let $c_1 = (\emptyset, \mathsf{None})$ represent \texttt{unknown}.
Let $c_2$ represent \texttt{null | undefined}: $c_2 = (\emptyset, \mathsf{Some}(\{0 \mapsto \texttt{undefined}, 2 \mapsto \texttt{null}\}))$.

\textbf{Incompleteness:}
By definition of $\diff_{\Clause}$:
\[
\diff_{\Clause}(c_1, c_2) = c_1 \quad \text{(since } \stypes_1 = \mathsf{None}\text{)}
\]
So $\sem{\diff_{\Clause}(c_1, c_2)} = \sem{c_1} = \Val$ (all values).

But the true set difference is:
\[
\sem{c_1} \setminus \sem{c_2} = \Val \setminus (\sem{\texttt{null}} \cup \sem{\texttt{undefined}}) \neq \Val
\]

The result is incomplete: $\diff$ returns \texttt{unknown} instead of the narrowed type 
\texttt{\{\}} (non-nullish values). This is because the clause diff cannot ``subtract from top''---it 
simply returns the left clause unchanged when $\stypes_1 = \mathsf{None}$.
