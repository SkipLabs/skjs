\author{}

\begin{abstract}
We formalize the internal type representation used by SKJS and study two related
type transformations:
(1) predicate-driven \emph{refinement} (control-flow narrowing), and
(2) the type-difference operation ($\diff$) used during type inference.
We state and prove soundness properties for these transformations, while
explicitly documenting the intended sources of incompleteness.
\end{abstract}

\subsection*{Type Hierarchy and Notation Summary}

\begin{keydef}
\textbf{Type representation hierarchy} (each level builds on the previous):
\begin{align*}
\underbrace{\SType}_{\text{single tag}}
&\;\xrightarrow{\;\text{set}\;}\;
\underbrace{\STypes}_{\text{tag-distinct}}
\;\xrightarrow{\;\text{+ params}\;}\;
\underbrace{\Clause}_{\clause{V}{S}} \\
&\;\xrightarrow{\;\text{map}\;}\;
\underbrace{\Clauses}_{V \mapsto \clause{V}{S}}
\;\xrightarrow{\;\text{+ insts}\;}\;
\underbrace{\TType}_{(I, C)}
\end{align*}

\medskip
\textbf{Semantic interpretation} $\sem{\cdot} : X \to \mathcal{P}(\Val)$:
\begin{align*}
\sem{\STypes} &= \textstyle\bigcup_{s \in S} \sem{s}
&\sem{\clause{V}{S}} &= \textstyle\left(\bigcap_{v \in V} \sem{v}\right) \cap \sem{S}
&\sem{(I,C)} &= \sem{I} \cup \sem{C}
\end{align*}

\medskip
\textbf{Soundness contracts}:
\begin{itemize}
\item \emph{Predicate soundness} (refinement): $\sem{t} \cap P \subseteq \sem{R(t)}$
\item \emph{Diff soundness} (P1+P2): $\sem{t_1}\setminus\sem{t_2} \subseteq \sem{\diff(t_1,t_2)} \subseteq \sem{t_1}$
\end{itemize}
\end{keydef}

\section{Runtime Model and Semantic Interface}

\begin{definition}[Runtime Tags]
\label{def:runtime-tags}
Let $\Tags$ be the set of runtime tags (the coarse runtime kind of a value). In the current implementation, tags are represented as integers, so one may take $\Tags = \{0,1,\ldots,10\}$.
\end{definition}

\begin{definition}[Runtime Values and Tags]
\label{def:runtime-values-tags}
Assume runtime values carry their tag as part of the representation.
We fix a family of payload sets $(\Payload_k)_{k \in \Tags}$ and define
\[
\Val \triangleq \{\, (k,p) \mid k \in \Tags \ \land\ p \in \Payload_k \,\}.
\]
Define the runtime tag function $\rtag : \Val \to \Tags$ as the first projection:
\[
\rtag((k,p)) = k.
\]
\paragraph{Concrete payloads (for this document).}
We make primitive payloads concrete and leave structured payloads abstract:
\begin{align*}
\Payload_0 &= \{\star\} & \Payload_1 &= \{\star\} & \Payload_2 &= \{\star\} \\
\Payload_3 &= \{\mathsf{true},\mathsf{false}\} & \Payload_4 &= \mathbb{R} & \Payload_5 &= \String \\
\Payload_6 &= \mathbb{Z} & \Payload_{10} &= \String \times \Payload_{10}'.
\end{align*}
Here $\Payload_{10}'$ is an arbitrary set (the constructor payload).
For the remaining tags $k \in \{7,8,9\}$ we leave $\Payload_k$ as arbitrary sets.
\end{definition}

\begin{definition}[Simple Types]
\label{def:simple-types}
Let $\SType$ be the set of \emph{simple types}, i.e. the atomic type forms that have a single runtime tag.
Formally, $\SType$ is the smallest set generated by the following constructors:
\begin{itemize}
\item primitive units: \texttt{undefined}, \texttt{void}, \texttt{null}
\item primitive values with optional finite literal-set payload:
  \begin{itemize}
  \item \(\texttt{boolean}(B)\) where \(B \subseteq_{\mathrm{fin}} \{\mathsf{true},\mathsf{false}\}\) or \(B=\top\),
  \item \(\texttt{number}(N)\) where \(N \subseteq_{\mathrm{fin}} \mathbb{R}\) or \(N=\top\),
  \item \(\texttt{string}(S)\) where \(S \subseteq_{\mathrm{fin}} \String\) or \(S=\top\),
  \item \(\texttt{bigint}(Z)\) where \(Z \subseteq_{\mathrm{fin}} \mathbb{Z}\) or \(Z=\top\).
  \end{itemize}
\item structured atoms: \texttt{tuple}$(\ldots)$, \texttt{Promise}$(\ldots)$, \texttt{object}$(\ldots)$, \texttt{algebraic}$(\ldots)$.
  These include tagged unions/ADTs with a runtime discriminant.
\end{itemize}
We write these constructors suggestively; most internal parameters are left abstract.
However, for control-flow refinement we will require a concrete semantic interface for object shapes (fields and index signatures); see Definition~\ref{def:object-semantics}.
\paragraph{Notation (source-level literal types).}
We write \texttt{boolean} for \(\texttt{boolean}(\top)\), and similarly for \texttt{number}, \texttt{string}, \texttt{bigint}.
We write \texttt{true} for \(\texttt{boolean}(\{\mathsf{true}\})\) and \texttt{false} for \(\texttt{boolean}(\{\mathsf{false}\})\).
We write a numeric literal \(n\) for \(\texttt{number}(\{n\})\), and a string literal ``\(x\)'' for \(\texttt{string}(\{x\})\).
More generally, a union like \texttt{1 | 2} is represented as \(\texttt{number}(\{1,2\})\), i.e.\ a single \(\SType\) alternative at the numeric tag.
We use these abbreviations uniformly, including under \(\tagfn(\cdot)\) and \(\sem{\cdot}\).
For example, \(\sem{\texttt{number}}\) means \(\sem{\texttt{number}(\top)}\), and \(\tagfn(\texttt{1})\) means \(\tagfn(\texttt{number}(\{1\}))\).
\end{definition}

\begin{definition}[Simple Types and Tags]
\label{def:simple-types-and-tags}
The function $\tagfn : \SType \to \Tags$ assigns tags:
\begin{align*}
\tagfn(\texttt{undefined}) &= 0 & \tagfn(\texttt{void}) &= 1 \\
\tagfn(\texttt{null}) &= 2 & \tagfn(\texttt{boolean}(\cdot)) &= 3 \\
\tagfn(\texttt{number}(\cdot)) &= 4 & \tagfn(\texttt{string}(\cdot)) &= 5 \\
\tagfn(\texttt{bigint}(\cdot)) &= 6 & \tagfn(\texttt{tuple}(\ldots)) &= 7 \\
\tagfn(\texttt{Promise}(\ldots)) &= 8 & \tagfn(\texttt{object}(\ldots)) &= 9 \\
\tagfn(\texttt{algebraic}(\ldots)) &= 10
\end{align*}
The semantic interpretation $\sem{\cdot} : \SType \to \mathcal{P}(\Val)$ maps each simple type to its set of inhabitants.
We define $\sem{\cdot}$ concretely for primitive units and primitive-value payloads (including singleton-literal notation), and define the remaining structured cases at tag level
(with the \texttt{object} and \texttt{algebraic} cases further constrained in Definitions~\ref{def:object-semantics} and~\ref{def:algebraic-semantics}):
\begin{align*}
\sem{\texttt{undefined}} &\triangleq \{(0,\star)\} &
\sem{\texttt{void}} &\triangleq \{(1,\star)\} &
\sem{\texttt{null}} &\triangleq \{(2,\star)\} \\[4pt]
\sem{\texttt{boolean}(\top)} &\triangleq \{(3,b) \mid b \in \{\mathsf{true},\mathsf{false}\}\} &
\sem{\texttt{boolean}(B)} &\triangleq \{(3,b) \mid b \in B\} \\
\sem{\texttt{number}(\top)} &\triangleq \{(4,n) \mid n \in \mathbb{R}\} &
\sem{\texttt{number}(N)} &\triangleq \{(4,n) \mid n \in N\} \\
\sem{\texttt{string}(\top)} &\triangleq \{(5,x) \mid x \in \String\} &
\sem{\texttt{string}(S)} &\triangleq \{(5,x) \mid x \in S\} \\
\sem{\texttt{bigint}(\top)} &\triangleq \{(6,z) \mid z \in \mathbb{Z}\} &
\sem{\texttt{bigint}(Z)} &\triangleq \{(6,z) \mid z \in Z\} \\[4pt]
\sem{\texttt{tuple}(\ldots)} &\triangleq \{ v \in \Val \mid \rtag(v)=7\} \\
\sem{\texttt{Promise}(\ldots)} &\triangleq \{ v \in \Val \mid \rtag(v)=8\}.
\end{align*}
In the finite-payload cases, $B$, $N$, $S$, $Z$ denote finite subsets of the respective domains.
\end{definition}

\begin{lemma}[\textbf{A0} (Runtime Tag Consistency)]
\label{ax:A0}
The semantic interpretation of simple types agrees with runtime tags:
\[
\forall s \in \SType.\; \forall v \in \Val.\; v \in \sem{s} \implies \rtag(v) = \tagfn(s).
\]
\end{lemma}

\begin{proof}
By case analysis on $s$.
For primitive units and primitive-value payloads, the definition of $\sem{s}$ contains only values of the form $(\tagfn(s),p)$, so $\rtag(v)=\tagfn(s)$.
For the tag-level structured cases (\texttt{tuple}, \texttt{Promise}), $\sem{s}$ is defined as $\{v \mid \rtag(v)=k\}$ where $k=\tagfn(s)$.
For \texttt{object} and \texttt{algebraic}, Definitions~\ref{def:object-semantics} and~\ref{def:algebraic-semantics} explicitly require membership in $\ObjVal$ (tag $9$) and $\AlgVal$ (tag $10$) respectively, and these tags agree with $\tagfn$.
\end{proof}

\begin{lemma}[\textbf{A1} (Tag Disjointness)]
\label{ax:A1}
\[
\forall s_1, s_2 \in \SType.\; \tagfn(s_1) \neq \tagfn(s_2) \implies \sem{s_1} \cap \sem{s_2} = \emptyset
\]
\end{lemma}

\begin{proof}
Let $k_1=\tagfn(s_1)$ and $k_2=\tagfn(s_2)$ with $k_1\neq k_2$.
By Lemma~\ref{ax:A0}, every $v \in \sem{s_1}$ satisfies $\rtag(v)=k_1$ and every $v \in \sem{s_2}$ satisfies $\rtag(v)=k_2$.
No value can have two different tags, so $\sem{s_1}\cap \sem{s_2}=\emptyset$.
\end{proof}

\begin{definition}[Object Values and Object-Type Semantics]
\label{def:object-semantics}
Let $\ObjVal \subseteq \Val$ be the set of object-tagged runtime values:
\[
\ObjVal \triangleq \{ v \in \Val \mid \rtag(v)=9\}.
\]
Assume a (partial) finite-property interface on objects:
\begin{align*}
\mathsf{props} &: \ObjVal \to \mathcal{P}_{\mathrm{fin}}(\String) \\
\mathsf{get} &: \ObjVal \times \String \rightharpoonup \Val
\end{align*}
such that $\mathsf{get}(o,p)$ is defined exactly when $p \in \mathsf{props}(o)$.
Write $\mathsf{has}(o,p)$ for $p \in \mathsf{props}(o)$.

\paragraph{Object field specifications.}
Assume predicates $\mathsf{callable}, \mathsf{constructible} : \ObjVal \to \{\mathsf{true},\mathsf{false}\}$ testing whether an object value is callable or constructible.

\paragraph{Three-valued signature modes.}
To support precise refinement after normalization, we use a three-valued mode for call and constructor signatures:
\[
\mathsf{SigMode} \triangleq \{\mathsf{None},\, \mathsf{Maybe}(\sigma),\, \mathsf{Definite}(\sigma)\}
\]
where $\sigma$ ranges over abstract signatures (we treat signatures opaquely in this document).
Intuitively:
\begin{itemize}
\item $\mathsf{Definite}(\sigma)$: the object is definitely callable/constructible with signature $\sigma$.
\item $\mathsf{Maybe}(\sigma)$: the object may or may not be callable/constructible; if it is, it has signature $\sigma$.
\item $\mathsf{None}$: the object is definitely not callable/constructible.
\end{itemize}

An \emph{object field specification} is a tuple $F=(F_{\mathrm{req}},F_{\mathrm{opt}},F_{\mathrm{idx}},F_{\mathrm{new}},F_{\mathrm{call}})$ where:
\begin{itemize}
\item $F_{\mathrm{req}} : \String \rightharpoonup \TType$ is a finite partial map of required fields,
\item $F_{\mathrm{opt}} : \String \rightharpoonup \TType$ is a finite partial map of optional fields,
\item $F_{\mathrm{idx}}$ is either $\mathsf{None}$ or $\mathsf{Some}(K,V)$ for $K,V \in \TType$ (an index signature),
\item $F_{\mathrm{new}} \in \mathsf{SigMode}$ for constructor signatures,
\item $F_{\mathrm{call}} \in \mathsf{SigMode}$ for call signatures.
\end{itemize}

\paragraph{Semantics of \texttt{object}$(F)$.}
We assume the object constructor in $\SType$ can carry such an $F$, and define:
\[
o \in \sem{\texttt{object}(F)}
\]
if and only if $o \in \ObjVal$ and the following constraints hold:
\begin{itemize}
\item \textbf{Required fields:} for every $p \in \mathrm{dom}(F_{\mathrm{req}})$,
$\mathsf{has}(o,p)$ and $\mathsf{get}(o,p) \in \sem{F_{\mathrm{req}}(p)}$.
\item \textbf{Optional fields:} for every $p \in \mathrm{dom}(F_{\mathrm{opt}})$,
if $\mathsf{has}(o,p)$ then $\mathsf{get}(o,p) \in \sem{F_{\mathrm{opt}}(p)}$ (and if not, no constraint).
\item \textbf{Index signature (open objects):} if $F_{\mathrm{idx}}=\mathsf{Some}(K,V)$ then for every $p \in \mathsf{props}(o)$:
if the runtime string value $\text{``}p\text{''}$ is in $\sem{K}$, then $\mathsf{get}(o,p) \in \sem{V}$.
\item \textbf{Constructor signature:}
\begin{itemize}
\item if $F_{\mathrm{new}} = \mathsf{Definite}(\nu)$ then $\mathsf{constructible}(o)=\mathsf{true}$;
\item if $F_{\mathrm{new}} = \mathsf{Maybe}(\nu)$ then no constraint (object may or may not be constructible);
\item if $F_{\mathrm{new}} = \mathsf{None}$ then $\mathsf{constructible}(o)=\mathsf{false}$.
\end{itemize}
\item \textbf{Call signature:}
\begin{itemize}
\item if $F_{\mathrm{call}} = \mathsf{Definite}(\kappa)$ then $\mathsf{callable}(o)=\mathsf{true}$;
\item if $F_{\mathrm{call}} = \mathsf{Maybe}(\kappa)$ then no constraint (object may or may not be callable);
\item if $F_{\mathrm{call}} = \mathsf{None}$ then $\mathsf{callable}(o)=\mathsf{false}$.
\end{itemize}
\end{itemize}
\paragraph{Note.}
This semantics is intentionally \emph{open}: objects may have properties not mentioned in $F_{\mathrm{req}}$ or $F_{\mathrm{opt}}$.
The three-valued signature modes ensure that normalization (merging callable with non-callable objects) is sound, while still permitting precise refinement via \texttt{typeof}.
\end{definition}

\begin{definition}[Algebraic Values and Tag-Based Algebraic-Type Semantics]
\label{def:algebraic-semantics}
Let $\AlgVal \subseteq \Val$ be the set of algebraic-tagged runtime values:
\[
\AlgVal \triangleq \{ v \in \Val \mid \rtag(v)=10\}.
\]
Recall that $\Payload_{10}=\String \times \Payload_{10}'$ (Definition~\ref{def:runtime-values-tags}).
Define the runtime discriminant function $\mathsf{atag} : \AlgVal \to \String$ by projection:
\[
\mathsf{atag}((10,(\ell,p))) \triangleq \ell.
\]
\paragraph{Semantics of \texttt{algebraic}$(T)$.}
We assume the algebraic constructor in $\SType$ can carry a finite set of tags $T \subseteq_{\mathrm{fin}} \String$,
and define:
\[
v \in \sem{\texttt{algebraic}(T)}
\quad\text{iff}\quad
v \in \AlgVal \ \land\  \mathsf{atag}(v) \in T.
\]
\paragraph{Note.}
This models only the discriminant-level behavior needed for refinement; field/constructor payload constraints are left abstract.
\end{definition}

\section{Internal Type Representation}

\begin{definition}[STypes]
\label{def:stypes}
\leavevmode
\begin{keydef}
\(\STypes\) is a finite set \(S \subseteq_{\mathrm{fin}} \SType\) with pairwise distinct tags.

\smallskip
\textbf{Semantics}: $\sem{S} = \bigcup_{s \in S} \sem{s}$.
\end{keydef}

\begin{enumerate}
\item \textbf{Tag-distinctness}: $s_1 \neq s_2 \implies \tagfn(s_1) \neq \tagfn(s_2)$.

\item \textbf{Lookup}: $\tags(S) = \{\tagfn(s) \mid s \in S\}$. Write $S[k]$ for the unique $s \in S$ with $\tagfn(s)=k$.

\item \textbf{Intuition}: An $\STypes$ value represents the \emph{tag-level} part of a union type---one representative $\SType$ per runtime tag.

\item \textbf{Example}: \texttt{number | string} is $\{\texttt{number}, \texttt{string}\}$.
  But \texttt{number} and literal \texttt{1} share tag $4$, so they cannot both appear in an $\STypes$.
\end{enumerate}
\end{definition}

\subsection{Normal-Form Operations on $\STypes$}
\label{sec:stypes-ops}

\paragraph{Operations (API).}
We treat \(\STypes\) as a normal-form ADT and highlight the operations used by the implementation:
\begin{itemize}
\item \textbf{Constructors:} \(\emptyset \in \STypes\) and \(\{s\} \in \STypes\) for \(s\in\SType\).
\item \textbf{Filter-map (refinement primitive):} for tag-preserving \(g\) and filter \(f\),
\[
\mathsf{filterMap}^{f,g}(S) \triangleq \{\, g(s) \mid s \in S,\ f(s)=\mathsf{true}\,\}.
\]
\item \textbf{Union (implementation union):} \(S_1 \sqcup_{\STypes} S_2\) is the normal-form union operation used by SKJS, which merges same-tag alternatives into a single representative per tag (while keeping other tags unchanged).
\item \textbf{Intersection (over-approximate meet):} the implementation provides a partial operation \(S_1 \sqcap_{\STypes} S_2\) (it may return no result when the intersection is bottom). We model it abstractly below via its soundness property.
\item \textbf{Difference (tag-elimination):} \(\diff_{\STypes}(S_1,S_2)\) is defined in Section~\ref{def:stypes-diff}.
\end{itemize}

\paragraph{Soundness summary.}
All operations above are intended to be sound with respect to \(\sem{\cdot}\):
\begin{itemize}
\item \textbf{Empty/singleton (exact):} \(\sem{\emptyset}=\emptyset\) and \(\sem{\{s\}}=\sem{s}\).
\item \textbf{Filter-map (predicate-sound):} if \((f,g)\) is sound for predicate \(P\) (Definition~\ref{def:filter-map-refinement}) and \(g\) preserves tags, then
\[
\sem{S} \cap P \subseteq \sem{\mathsf{filterMap}^{f,g}(S)}.
\]
\item \textbf{Union (over-approximate union):}
\[
\sem{S_1} \cup \sem{S_2} \subseteq \sem{S_1 \sqcup_{\STypes} S_2}.
\]
\item \textbf{Intersection (over-approximate intersection):} when \(S_1 \sqcap_{\STypes} S_2\) is defined,
\[
\sem{S_1} \cap \sem{S_2} \subseteq \sem{S_1 \sqcap_{\STypes} S_2}.
\]
\item \textbf{Difference:} (P1) always holds (Lemma~\ref{lem:stypes-diff-p1}); (P2) holds under \(\mathsf{STC}\) (Lemma~\ref{lem:stypes-diff-p2-under-stc}).
\end{itemize}

\begin{lemma}[Union Soundness for \(\STypes\)]
\label{lem:stypes-union-sound}
For all \(S_1,S_2 \in \STypes\),
\[
\sem{S_1} \cup \sem{S_2} \subseteq \sem{S_1 \sqcup_{\STypes} S_2}.
\]
\end{lemma}

\begin{proof}
We will define \(S_1 \sqcup_{\STypes} S_2\) to match the implementation's tag-directed union operation (Definition~\ref{def:stypes-union-inter}), and soundness follows from the corresponding same-tag merge soundness of each case.
\end{proof}

\begin{definition}[Implementation Union/Intersection on $\STypes$]
\label{def:stypes-union-inter}
We define \(\sqcup_{\STypes}\) and \(\sqcap_{\STypes}\) to model SKJS's operations on normal-form tag-distinct unions.

\paragraph{Same-tag merge on simple types.}
Define partial operators \(\sqcup_{\SType}\) and \(\sqcap_{\SType}\) on \(\SType\), used only when \(\tagfn(s_1)=\tagfn(s_2)\).
They follow the implementation's tag-dispatch (see \texttt{STypes::union} and \texttt{STypes::inter}):
\begin{itemize}
\item \textbf{Primitive units (tags 0/1/2):}
  \(s_1 \sqcup_{\SType} s_2 \triangleq s_1\) and \(s_1 \sqcap_{\SType} s_2 \triangleq s_1\).
  The implementation keeps the left operand; semantics is unchanged.

\item \textbf{Primitive values (tags 3/4/5/6):}
  \(\sqcup_{\SType}\) computes the set union of literal-sets when both are finite, and otherwise yields the full primitive kind (e.g.\ if either operand is \(\top\), the result is \(\top\)).
  For booleans, unioning \(\{\mathsf{true}\}\) and \(\{\mathsf{false}\}\) yields \(\texttt{boolean}(\top)\).
  Intersection \(\sqcap_{\SType}\) keeps only common literals, or is undefined if none.

\item \textbf{Tuples (tag 7):}
  \(\sqcup_{\SType}\) unions element-wise up to the shorter length; missing positions in the shorter tuple are treated as \texttt{undefined} and unioned with the corresponding element of the longer tuple.
  \(\sqcap_{\SType}\) intersects element-wise up to the shorter length and appends the remainder of the longer tuple unchanged.

\item \textbf{Promises (tag 8):}
  \(\texttt{Promise}(t_1) \sqcup_{\SType} \texttt{Promise}(t_2) \triangleq \texttt{Promise}(t_1 \sqcup_{\TType} t_2)\), and similarly \(\sqcap_{\SType}\) uses \(\sqcap_{\TType}\) (left abstract here).

\item \textbf{Objects (tag 9):}
  For \(\texttt{object}(F_1) \sqcup_{\SType} \texttt{object}(F_2)\):
  \begin{itemize}
  \item \emph{Fields:} union shared field types; make non-shared fields optional.
  \item \emph{Index signatures:} union key/value types into a single index signature.
  \item \emph{Call/new signatures:} keep only if both sides have one; otherwise drop.
  \end{itemize}
  Intersection is defined analogously and may be undefined (bottom) in some cases.

\item \textbf{Algebraic (tag 10):}
  \(\texttt{algebraic}(T_1) \sqcup_{\SType} \texttt{algebraic}(T_2)\) unions the constructor-tag set; when a constructor exists on both sides, it merges its field specifications using object-field union.
  \(\sqcap_{\SType}\) intersects constructor-tag sets and merges shared constructors; it is undefined if the tag-set intersection is empty.
\end{itemize}

\paragraph{Union on \(\STypes\).}
For \(S_1,S_2 \in \STypes\), define:
\begin{align*}
S_1 \sqcup_{\STypes} S_2 \triangleq{} &
  \{\, S_1[k] \mid k \in \tags(S_1)\setminus \tags(S_2)\,\} \\
  {}\cup{} & \{\, S_2[k] \mid k \in \tags(S_2)\setminus \tags(S_1)\,\} \\
  {}\cup{} & \{\, S_1[k] \sqcup_{\SType} S_2[k] \mid k \in \tags(S_1)\cap \tags(S_2)\,\}.
\end{align*}

\paragraph{Intersection on \(\STypes\).}
Define the (partial) intersection:
\begin{align*}
S_1 \sqcap_{\STypes} S_2 \triangleq{} &
  \{\, S_1[k] \sqcap_{\SType} S_2[k] \mid {} \\
  & \quad k \in \tags(S_1)\cap \tags(S_2) \text{ and } S_1[k]\sqcap_{\SType}S_2[k]\text{ defined}\,\}.
\end{align*}
If this set is empty, we treat the result as undefined (bottom), matching the implementation behavior.
\end{definition}

\begin{lemma}[Soundness of \(\sqcup_{\STypes}\) and \(\sqcap_{\STypes}\)]
\label{lem:stypes-union-inter-sound}
For all \(S_1,S_2 \in \STypes\):
\[
\sem{S_1} \cup \sem{S_2} \subseteq \sem{S_1 \sqcup_{\STypes} S_2}.
\]
Moreover, when \(S_1 \sqcap_{\STypes} S_2\) is defined,
\[
\sem{S_1} \cap \sem{S_2} \subseteq \sem{S_1 \sqcap_{\STypes} S_2}.
\]
\end{lemma}

\begin{proof}
For \(\sqcup_{\STypes}\): let \(v \in \sem{S_1} \cup \sem{S_2}\). WLOG \(v \in \sem{S_1}\), so \(v \in \sem{S_1[k]}\) for some \(k \in \tags(S_1)\).
If \(k \notin \tags(S_2)\), then \(S_1[k] \in S_1 \sqcup_{\STypes} S_2\), hence \(v \in \sem{S_1 \sqcup_{\STypes} S_2}\).
If \(k \in \tags(S_2)\), then \(S_1[k]\sqcup_{\SType}S_2[k]\in S_1\sqcup_{\STypes}S_2\), and inclusion reduces to the same-tag merge soundness of \(\sqcup_{\SType}\) for that tag case.

For \(\sqcap_{\STypes}\): let \(v \in \sem{S_1}\cap \sem{S_2}\). Then \(v \in \sem{S_1[k]}\) and \(v \in \sem{S_2[k]}\) for the same tag \(k\) by A1, and inclusion reduces to the same-tag merge soundness of \(\sqcap_{\SType}\) when defined.
\end{proof}

\subsection{Completeness and Precision Loss in Normal-Form Operations}
\label{sec:stypes-precision-loss}

We say an operation is \emph{complete} when its result is semantically exact (i.e.\ no extra values are introduced), and \emph{imprecise} when it introduces a strict over-approximation.
This section identifies the specific implementation-driven steps where over-approximation is introduced, to inform potential future precision warnings.

\paragraph{Union is exact for primitive tags.}
For primitive units and primitive values, \(\sqcup_{\STypes}\) is complete: it computes exactly the semantic union of the two alternatives at those tags.
(E.g.\ numeric/string/bigint literal-sets are explicitly stored and unioned; boolean literals union to \texttt{boolean}.)

\paragraph{Union is generally imprecise for structured tags.}
For tuples, objects, promises, and algebraics, the same-tag merge rules may introduce extra inhabitants.
Key sources of imprecision:
\begin{itemize}
\item \textbf{Object field merging via optionalization:}
  Merging \(\texttt{object}(\{a:\tau\})\) and \(\texttt{object}(\{b:\upsilon\})\) yields an object type where both \(a\) and \(b\) are optional.
  This admits objects containing \emph{both} fields, which need not be in the original disjoint union.
  This is the primary, intentional precision loss caused by \(\STypes\) storing at most one object alternative per tag.

\item \textbf{Index signature union:}
  When both sides have index signatures, SKJS unions key/value types into a single index signature, losing per-key distinctions (see \texttt{TIndexField::union}).

\item \textbf{Call/new signature merge:}
  When only one side has a call (or new) signature, SKJS drops it in the merged object.
  This loses the ability to distinguish callable from non-callable alternatives inside a single object tag.

\item \textbf{Tuple union padding with \texttt{undefined}:}
  When tuple lengths differ, missing elements in the shorter tuple are unioned with \texttt{undefined}, potentially admitting tuples with explicit \texttt{undefined} entries not present in the shorter alternative.
\end{itemize}

\paragraph{Intersection imprecision for structured tags.}
As with union, \(\sqcap_{\STypes}\) is exact for primitive units and primitive values (it intersects literal-sets precisely when they are tracked), but it can introduce over-approximation for structured tags:
\begin{itemize}
\item \textbf{Index signature intersection uses union (over-approximation):}
  When intersecting two index signatures, the implementation returns a \emph{single} index signature whose key and value types are \emph{unioned} (see \texttt{TIndexField::inter}).
  This is sound but can admit extra key/value pairs that satisfy neither original signature.

\item \textbf{Tuple intersection on unequal lengths:}
  The implementation intersects element-wise up to the shorter length and appends the remaining elements of the longer tuple unchanged.
  This can admit tuples not in the exact set-theoretic intersection.
\end{itemize}

\paragraph{Clause-level intersection imprecision.}
At the clause level (not \(\STypes\)), \texttt{Clauses::inter} accumulates clause-pair intersections in a map keyed by the variable-set.
When multiple intersections land on the same key, they are combined using \texttt{Clause::union}, widening the result beyond the exact intersection in exchange for keeping at most one clause per key.

\paragraph{Difference is intentionally incomplete at same-tag.}
\(\diff_{\STypes}\) removes \emph{whole tags}. Therefore it cannot express same-tag subtraction such as \(\texttt{number} \setminus \texttt{1}\), which is why (P2) requires the same-tag coverage side condition \(\mathsf{STC}\) (Section~\ref{def:stc}).

\paragraph{Filter-map precision loss.}
\(\mathsf{filterMap}^{f,g}\) can lose precision whenever the filter drops alternatives or whenever the mapper performs a conservative rewrite. In SKJS, this shows up in refinements that (a) retain extra tags conservatively (e.g. \texttt{"in"} keeps tuples/promises/algebraics), or (b) perform only tag-level filtering without tracking richer correlations between fields.

\begin{definition}[Type Instances]
\label{def:type-instances}
Let $\TInst$ be the set of type instances (type variables or generic instantiations).
\paragraph{Examples.}
A type variable instance is something like $T$ (a unification variable introduced during type checking; this is \emph{not} a source-level type parameter).
A generic instantiation instance is a named type applied to type arguments, e.g. $\texttt{Promise}\langle \texttt{number} \rangle$ or $\texttt{Array}\langle \texttt{string} \rangle$.

Each instance $i \in \TInst$ has a semantic interpretation $\sem{i} \subseteq \Val$.
\end{definition}

\begin{definition}[Type Instance Sets]
\label{def:type-instance-sets}
Let $\TInsts = \mathcal{P}_{\mathrm{fin}}(\TInst)$ be the set of \emph{finite} sets of instances.
For $I \in \TInsts$:
\[
\sem{I} = \bigcup_{i \in I} \sem{i}.
\]
\end{definition}

\begin{definition}[Type Parameters]
\label{def:type-parameters}
Let $\TParam$ be the set of \emph{type parameters} (generic binders), i.e.\ names introduced by source-level generic declarations such as $\texttt{function f}\langle T\rangle(\ldots)$ or $\texttt{type Box}\langle T\rangle = \ldots$.
Type parameters differ from type-variable instances: parameters are binders whose meaning is fixed by the caller, whereas instances are inference unknowns solved by the type checker.
\end{definition}

\begin{definition}[Clause and Clauses]
\label{def:clause-clauses}
\leavevmode
\begin{keydef}
A \emph{clause} $\clause{V}{S}$ pairs type parameters $V \subseteq \TParam$ with $S \in \STypes \cup \{\top\}$.

\smallskip
$\Clauses$ is a map from finite subsets of $\TParam$ to clauses, at most one per key.
\end{keydef}

\begin{enumerate}
\item \textbf{Notation}: We write $\clause{V}{S}$ for a clause with parameters $V$ and simple-types $S$.
  The symbol $\top$ denotes the ``top'' case (no type constraint); $\clause{\emptyset}{\top}$ is the unique top clause.

\item \textbf{Semantics}:
\begin{align*}
\sem{\clause{V}{S}} &= \textstyle\left(\bigcap_{v \in V} \sem{v}\right) \cap \sem{S}
& \text{where } \sem{\top} = \Val \\
\sem{C} &= \textstyle\bigcup_{k \in \mathrm{dom}(C)} \sem{C(k)}
\end{align*}

\item \textbf{Well-formedness}: Keys agree with clause parameters: $C(k) = \clause{V}{S} \implies V = k$.
  Top implies empty parameters: $S = \top \implies V = \emptyset$.

\item \textbf{Example}: Let $A, B \in \TParam$.
  The clauses map $C = \{A \mapsto \clause{\{A\}}{\texttt{number}},\; B \mapsto \clause{\{B\}}{\texttt{string}}\}$ has:
\[
\sem{C} = (\sem{A} \cap \sem{\texttt{number}}) \cup (\sem{B} \cap \sem{\texttt{string}}).
\]
A clause $\clause{\{A,B\}}{\texttt{number}}$ requires \emph{both} parameters: $\sem{A} \cap \sem{B} \cap \sem{\texttt{number}}$.
\end{enumerate}
\end{definition}

\begin{definition}[TType]
\label{def:ttype}
$\TType = (\insts, \clauses)$ where $\insts \in \TInsts$ and $\clauses \in \Clauses$.
\[
\sem{(\insts, \clauses)} = \sem{\insts} \cup \sem{\clauses}
\]
\paragraph{Schematic.}
It is helpful to keep the following picture in mind:
\[
\underbrace{(\insts,\clauses)}_{\TType}
\quad=\quad
\underbrace{\insts}_{\text{``instance'' part}}
\;\;\cup\;\;
\underbrace{\clauses}_{\text{``conditional'' part}}
\]
and later $\diff_{\TType}$ simply applies $\diff$ componentwise to these two parts.
\paragraph{Example.}
Let $I = \{\texttt{Promise}\langle \texttt{number} \rangle\} \in \TInsts$ and let $C$ be the clauses map from the previous example.
Then $t = (I,C)$ denotes the type whose semantics is:
\[
\sem{t} = \sem{\texttt{Promise}\langle \texttt{number} \rangle} \cup \sem{C}.
\]
\end{definition}

\section{Subtyping (Skeleton)}
\label{sec:subtyping-skeleton}

We introduce a subtyping judgement on internal types,
written $t_1 \leq t_2$ for $t_1,t_2 \in \TType$.
In this document, $\leq$ is intended to be a faithful mathematical model of the
subtyping \emph{validator} implemented by SKJS (the accept/reject check used by the type checker),
rather than an abstract, complete, semantic inclusion relation.

\paragraph{Soundness theorem (to be proved).}
The central meta-theoretic property we ultimately want from the validator-model judgement $\leq$
is semantic soundness:
\[
t_1 \leq t_2 \quad\Longrightarrow\quad \sem{t_1} \subseteq \sem{t_2}.
\]
In a full development, this is proved from the subtyping rules/algorithm that define $\leq$.

\paragraph{Incompleteness.}
The converse implication is intentionally \emph{not} assumed: SKJS may reject some valid
semantic inclusions, so $\leq$ can be incomplete.

\paragraph{Note (expansion and substitution effects).}
In the implementation, the validator operates on expanded/normalized views of types using the current substitution
and a variance parameter, and it may update the substitution while checking (as part of inference).
We do not model these mechanisms explicitly here; they are absorbed into the meaning of the judgement $t_1 \leq t_2$.

\paragraph{Example obligations (singleton literal types).}
Once $\leq$ is defined by the validator's rules, it should validate the expected singleton-to-primitive
subtypings for primitive literals (these are representative, actually-implemented cases).
(In examples we write \texttt{1}, \texttt{"x"}, \texttt{number}, \texttt{string} for the corresponding internal types,
by abuse of notation.) For example:
\[
\texttt{1} \leq \texttt{number}
\qquad
\texttt{number} \not\leq \texttt{1}.
\]
Similarly, a string literal is a subtype of \texttt{string}:
\[
\texttt{"x"} \leq \texttt{string}.
\]

\subsection{Soundness Framework}
\label{sec:refine-framework}

\begin{definition}[Refinement Judgements and Soundness]
\label{def:refinement}
Let $\Env$ map local variables to $\TType$; let $\sigma$ map locals to $\Val$.
Write $\sigma \models \Env$ when $\sigma(x) \in \sem{\Env(x)}$ for all $x$.

The judgements $\RefineT{\Env}{e}{\Env'}$ and $\RefineF{\Env}{e}{\Env'}$ refine $\Env$ under the assumption that $e$ is true/false.
The system is \emph{branch-sound} if: whenever $\sigma \models \Env$ and $e$ evaluates to $b \in \{\mathsf{true}, \mathsf{false}\}$, then $\sigma \models \Env'$ for the corresponding derivation.
\end{definition}

\begin{definition}[\texttt{typeof} Predicates]
\label{def:typeof-predicates}
Using the $\mathsf{callable}$ predicate from Definition~\ref{def:object-semantics}, define the following predicates:
\begin{align*}
P_{\mathsf{typeof}=\texttt{"string"}} &\triangleq \{ v \in \Val \mid \rtag(v)=5 \} \\
P_{\mathsf{typeof}=\texttt{"number"}} &\triangleq \{ v \in \Val \mid \rtag(v)=4 \} \\
P_{\mathsf{typeof}=\texttt{"bigint"}} &\triangleq \{ v \in \Val \mid \rtag(v)=6 \} \\
P_{\mathsf{typeof}=\texttt{"boolean"}} &\triangleq \{ v \in \Val \mid \rtag(v)=3 \} \\
P_{\mathsf{typeof}=\texttt{"function"}} &\triangleq \{ o \in \ObjVal \mid \mathsf{callable}(o) \} \\
P_{\mathsf{typeof}=\texttt{"undefined"}} &\triangleq \{ v \in \Val \mid \rtag(v)\in\{0,1\} \} \\
P_{\mathsf{typeof}=\texttt{"object"}} &\triangleq \ObjVal \cup \AlgVal \cup \sem{\texttt{null}}.
\end{align*}
\paragraph{Note.}
The \texttt{"symbol"} case is intentionally omitted because SKJS performs no refinement for it.
\end{definition}

\begin{definition}[Evaluation for Refinement Tests]
\label{def:refine-eval}
We define a partial evaluation relation for the refinement-test fragment:
\[
\sigma \models_{\mathsf{true}} e
\qquad\text{and}\qquad
\sigma \models_{\mathsf{false}} e.
\]
It is defined only for the supported test forms:
\begin{itemize}
\item \textbf{Nullish:} $\sigma \models_{\mathsf{true}} (x===\texttt{null})$ iff $\rtag(\sigma(x))=2$; $\sigma \models_{\mathsf{false}}$ is the negation.
Similarly for \texttt{undefined}, with $\rtag(\sigma(x))\in\{0,1\}$.
\item \textbf{\texttt{typeof}:} $\sigma \models_{\mathsf{true}} (\texttt{typeof}\ x===k)$ iff $\sigma(x)\in P_{\mathsf{typeof}=k}$ where $P_{\mathsf{typeof}=k}$ is as in Definition~\ref{def:typeof-predicates}; the false relation is negation.
\item \textbf{\texttt{in}:} $\sigma \models_{\mathsf{true}} (\texttt{"}p\texttt{" in }x)$ iff $\sigma(x)\in\ObjVal$ and $\mathsf{has}(\sigma(x),p)$; the false relation is negation.
\item \textbf{Algebraic tag equality:} $\sigma \models_{\mathsf{true}} (x.\texttt{\_tag}===\ell)$ iff $\sigma(x)\in\AlgVal$ and $\mathsf{atag}(\sigma(x))=\ell$, where $\ell \in \String$ is a string literal; the false relation is negation.
\item \textbf{\texttt{instanceof}:} $\sigma \models_{\mathsf{true}} (x\ \texttt{instanceof}\ C)$ iff $\sigma(x)\in P_{\mathsf{inst}}(C)$ for a predicate $P_{\mathsf{inst}}(C)\subseteq \Val$ (see Definition~\ref{def:instanceof-semantics}); the false relation is negation.
\item \textbf{Connectives:}
  $\sigma \models_{\mathsf{true}} (e_1 \ \&\&\ e_2)$ iff $\sigma \models_{\mathsf{true}} e_1$ and $\sigma \models_{\mathsf{true}} e_2$.
  $\sigma \models_{\mathsf{false}} (e_1 \ ||\ e_2)$ iff $\sigma \models_{\mathsf{false}} e_1$ and $\sigma \models_{\mathsf{false}} e_2$.
  (Other combinations are omitted, since SKJS only uses these connective cases for refinement in the current implementation.)
\end{itemize}
\end{definition}

\begin{definition}[Filter/Map Refinement (Implementation Model)]
\label{def:filter-map-refinement}
Let $P \in \Pred$ be a predicate.
Let $f : \SType \to \{\mathsf{true},\mathsf{false}\}$ be a filter and $g : \SType \to \SType$ be a mapper.
We say $(f,g)$ is \emph{sound for $P$} if for every $s \in \SType$ and every $v \in \sem{s}$:
\[
v \in P \implies (f(s) = \mathsf{true} \ \land\  v \in \sem{g(s)}).
\]
\paragraph{Remark (tag preservation).}
In the implementation, the mapper $g$ used for refinement preserves runtime tags:
$\tagfn(g(s))=\tagfn(s)$.
In this document we assume tag preservation when defining $\refineop_{\STypes}^{f,g}$, so that the result is again a well-formed $\STypes$ set (pairwise distinct tags). This matches SKJS's \texttt{FilterType} implementation.
\paragraph{Lift to $\STypes$.}
Define:
\[
\refineop_{\STypes}^{f,g}(S) \triangleq \{\, g(s) \mid s \in S,\ f(s)=\mathsf{true}\,\}.
\]
Operationally, this is exactly the ``filter-map'' over the unique representative per tag.
\paragraph{Lift to $\Clause$.}
For a clause $c = \clause{V}{S}$ define:
\[
\refineop_{\Clause}^{f,g}(\clause{V}{S}) \triangleq
\begin{cases}
\clause{V}{\top} & \text{if } S = \top \\
\clause{V}{\refineop_{\STypes}^{f,g}(S)} & \text{otherwise}
\end{cases}
\]
If the result is $\clause{V}{\emptyset}$, treat it as $\bot$ (dropped).

\paragraph{Lift to $\Clauses$ and $\TType$.}
Apply $\refineop_{\Clause}^{f,g}$ to each clause, dropping $\bot$ results.
For $t=(I,C)$: $\refineop^{f,g}(t) \triangleq (I,\refineop_{\Clauses}^{f,g}(C))$.
\end{definition}

\begin{definition}[Defaulting Refinement (Model of \texttt{setLocal})]
\label{def:defaulting-refinement}
Some refinements in SKJS provide a \emph{default} type when the current type is too imprecise to refine (notably when it is top or has no informative alternatives).
This is modeled by a \emph{defaulting refinement} operator.

Let $R : \TType \to \TType$ be a transformer and let $d$ be an \emph{optional} default simple type ($d \in \{\mathsf{None}\} \cup \{\mathsf{Some}(s)\mid s\in \SType\}$).
\paragraph{Singleton constructor.}
Write $\mathsf{single}(s)$ for the $\TType$ value consisting of a single clause with $\STypes=\{s\}$ and no instances.
\paragraph{Top/empty test.}
We say a $\TType$ value is \emph{syntactically top} if its clause component is $\clause{\emptyset}{\top}$,
and \emph{syntactically empty} if its clause component is empty (bottom).
\paragraph{Definition.}
Define $\mathsf{def}(R,d) : \TType \to \TType$ by:
\[
\mathsf{def}(R,d)(t) \triangleq
\begin{cases}
R(t) & \text{if } d=\mathsf{None} \\
\mathsf{single}(s) & \text{if } d=\mathsf{Some}(s) \text{ and } R(t) \text{ is syntactically top or empty} \\
R(t) & \text{otherwise}
\end{cases}
\]
\paragraph{Note.}
This abstracts the implementation check in \texttt{setLocal} based on \texttt{getSTypes} (after expansion): it injects a default when refinement would otherwise leave a top/empty shape.
\end{definition}

\begin{definition}[Predicate-Sound Type Transformer]
\label{def:predicate-sound-transformer}
Let $P \in \Pred$ and let $R : \TType \to \TType$.
We say $R$ is \emph{predicate-sound for $P$} if for all $t \in \TType$:
\[
\sem{t} \cap P \subseteq \sem{R(t)}.
\]
\end{definition}

\begin{lemma}[Filter/Map Refinement is Predicate-Sound]
\label{lem:filter-map-predicate-sound}
If $(f,g)$ is sound for $P$, then the transformer $t \mapsto \refineop^{f,g}(t)$ is predicate-sound for $P$ (Definition~\ref{def:predicate-sound-transformer}).
\end{lemma}

\begin{proof}
Let $t=(\insts,\clauses)$.
We prove $\sem{t}\cap P \subseteq \sem{\refineop^{f,g}(t)}$ by cases on membership in $\sem{\insts} \cup \sem{\clauses}$.
\begin{itemize}
\item If $v \in \sem{\insts}\cap P$, then $v \in \sem{\insts} \subseteq \sem{\refineop^{f,g}(t)}$ since refinement leaves $\insts$ unchanged.
\item If $v \in \sem{\clauses}\cap P$, then $v \in \sem{\clause{V}{S}}$ for some clause in $\clauses$.
By clause semantics, $v \in \bigcap_{w \in V} \sem{w}$ and $v \in \sem{S}$.
If $S=\top$, refinement leaves the clause unchanged and $v$ is still included.
Otherwise $v \in \sem{s}$ for some $s \in S$; by soundness of $(f,g)$, $f(s)=\mathsf{true}$ and $v\in \sem{g(s)}$.
Hence $v \in \sem{\refineop_{\STypes}^{f,g}(S)}$ and $v \in \sem{\clause{V}{\refineop_{\STypes}^{f,g}(S)}}$.
\end{itemize}
Combining cases yields the desired inclusion.
\end{proof}

\begin{lemma}[Defaulting Preserves Predicate-Soundness]
\label{lem:defaulting-preserves-predicate-sound}
Let $R:\TType \to \TType$ be predicate-sound for $P \subseteq \Val$.
Let $d$ be an optional default.
If $d=\mathsf{None}$, then $\mathsf{def}(R,d)$ is predicate-sound for $P$.
If $d=\mathsf{Some}(s)$ and $P \subseteq \sem{\mathsf{single}(s)}$, then $\mathsf{def}(R,d)$ is predicate-sound for $P$.
\end{lemma}

\begin{proof}
If $d=\mathsf{None}$, then $\mathsf{def}(R,d)=R$ by Definition~\ref{def:defaulting-refinement}, so predicate-soundness is immediate.
If $d=\mathsf{Some}(s)$, let $t \in \TType$ and let $v \in \sem{t}\cap P$.
If $R(t)$ is not syntactically top/empty, then $\mathsf{def}(R,d)(t)=R(t)$ and $v \in \sem{R(t)}$ by predicate-soundness of $R$.
If $R(t)$ is syntactically top/empty, then $\mathsf{def}(R,d)(t)=\mathsf{single}(s)$, and by hypothesis $v \in P \subseteq \sem{\mathsf{single}(s)}$.
Thus $\sem{t}\cap P \subseteq \sem{\mathsf{def}(R,d)(t)}$ for all $t$.
\end{proof}

\subsection{Inference Rules}
\label{sec:refine-rules}

\begin{keydef}
\textbf{Filter/Mapper summary.} Each test form uses filter $f$, mapper $g$, optional default $d$.
True branch: $\mathsf{def}(\refineop^{f,g}, d)$; false branch: $\refineop^{f',g'}$.

\smallskip
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}l@{\;\;}l@{\;\;}l@{\;\;}l@{}}
\toprule
\textbf{Test} & \textbf{Filter (true)} & \textbf{Mapper} & \textbf{Default} \\
\midrule
\texttt{x === null} & $\tagfn(s)=2$ & identity & --- \\
\texttt{x === undefined} & $\tagfn(s)\in\{0,1\}$ & identity & --- \\
\texttt{typeof x === "string"} & $\tagfn(s)=5$ & identity & \texttt{string} \\
\texttt{typeof x === "number"} & $\tagfn(s)=4$ & identity & \texttt{number} \\
\texttt{typeof x === "function"} & callable object & $\mathsf{Maybe}\to\mathsf{Definite}$ & --- \\
\texttt{typeof x === "object"} & $\tagfn(s)\in\{2,9,10\}$ & identity & \texttt{object}$(\emptyset)$ \\
\texttt{"p" in x} & tags 7--10 & add required $p$ & \texttt{object}$(\{p\})$ \\
\texttt{x.\_tag === $\ell$} & always true & filter to $\{\ell\}$ & --- \\
\texttt{x instanceof C} & (overwrite) & $\to T_{\mathsf{inst}}(C)$ & --- \\
\bottomrule
\end{tabular}
\end{keydef}

\paragraph{Local update.}
$\mathsf{upd}(\Env,x,R) \triangleq \Env[x \mapsto R(\Env(x))]$.
If no rule applies, refinement leaves $\Env$ unchanged.
Strict inequality \texttt{!==} swaps true/false branches.

\paragraph{Nullish filters and mappers.}
\begin{definition}[Nullish Filters and Mappers]
\label{def:nullish-filters-mappers}
Define the following filter/map functions on simple types:
\[
g_{\mathrm{id}}(s) \triangleq s.
\]
\[
f_{\mathsf{null}}(s) \triangleq (\tagfn(s)=2),
\qquad
f_{\neg\mathsf{null}}(s) \triangleq (\tagfn(s)\neq 2).
\]
\[
f_{\mathsf{undef}}(s) \triangleq (\tagfn(s)\in\{0,1\}),
\qquad
f_{\neg\mathsf{undef}}(s) \triangleq (\tagfn(s)\notin\{0,1\}).
\]
\end{definition}

\paragraph{Nullish equality.}
\begin{mathpar}
\inferrule*[right=R-NullT]{ }{\RefineT{\Env}{(x === \texttt{null})}{\mathsf{upd}(\Env,x,\,t \mapsto \refineop^{f_{\mathsf{null}},g_{\mathrm{id}}}(t))}}
\and
\inferrule*[right=R-NullF]{ }{\RefineF{\Env}{(x === \texttt{null})}{\mathsf{upd}(\Env,x,\,t \mapsto \refineop^{f_{\neg\mathsf{null}},g_{\mathrm{id}}}(t))}}
\and
\inferrule*[right=R-UndefT]{ }{\RefineT{\Env}{(x === \texttt{undefined})}{\mathsf{upd}(\Env,x,\,t \mapsto \refineop^{f_{\mathsf{undef}},g_{\mathrm{id}}}(t))}}
\and
\inferrule*[right=R-UndefF]{ }{\RefineF{\Env}{(x === \texttt{undefined})}{\mathsf{upd}(\Env,x,\,t \mapsto \refineop^{f_{\neg\mathsf{undef}},g_{\mathrm{id}}}(t))}}
\end{mathpar}
where $f_{\mathsf{null}}, f_{\neg\mathsf{null}}, f_{\mathsf{undef}}, f_{\neg\mathsf{undef}}, g_{\mathrm{id}}$ are as in Definition~\ref{def:nullish-filters-mappers}.
\paragraph{Side condition (locals only).}
In the implementation, comparisons against \texttt{null}/\texttt{undefined} are only permitted when the compared expression is a local variable; otherwise a typing error is raised.
Accordingly, the rules above are intended only for tests where $x$ is a local.

\paragraph{\texttt{typeof} filters and defaults.}
\begin{definition}[\texttt{typeof} Filters and Defaults]
\label{def:typeof-filters-defaults}
For each $k$ in Definition~\ref{def:typeof-predicates}, we define a filter/map pair and an optional default $d_k$ matching the implementation.
\begin{itemize}
\item For $k \in \{\texttt{"string"},\texttt{"number"},\texttt{"bigint"},\texttt{"boolean"},\texttt{"undefined"}\}$:
  $f_k(s)$ is true iff $\tagfn(s)$ is the corresponding tag, and $g_k \triangleq g_{\mathrm{id}}$.

\item For $k=\texttt{"object"}$:
  $f_{\texttt{"object"}}(s)$ is true iff $\tagfn(s)\in\{2,9,10\}$ (object, null, or algebraic), and $g_{\texttt{"object"}} \triangleq g_{\mathrm{id}}$.

\item For $k=\texttt{"function"}$:
  \begin{itemize}
  \item $f_{\texttt{"function"}}(s) = \mathsf{true}$ iff $s=\texttt{object}(F)$ with $F_{\mathrm{call}} \in \{\mathsf{Definite}(\kappa), \mathsf{Maybe}(\kappa)\}$.
  \item $g_{\texttt{"function"}}(s)$: if $F_{\mathrm{call}} = \mathsf{Maybe}(\kappa)$, return $\texttt{object}(F')$ with $F'_{\mathrm{call}} = \mathsf{Definite}(\kappa)$; otherwise return $s$.
  \end{itemize}
\end{itemize}
\paragraph{False-branch filter and mapper for \texttt{"function"}.}
For the false branch of \texttt{typeof x === "function"}, we use:
\begin{itemize}
\item $f_{\neg\texttt{"function"}}(s) = \mathsf{true}$ iff $s$ is not an object, or $s=\texttt{object}(F)$ with $F_{\mathrm{call}} \in \{\mathsf{None}, \mathsf{Maybe}(\kappa)\}$.
\item $g_{\neg\texttt{"function"}}(s)$: if $F_{\mathrm{call}} = \mathsf{Maybe}(\kappa)$, set $F'_{\mathrm{call}} = \mathsf{None}$; else return $s$.
\end{itemize}
\paragraph{Note.}
The key insight is that $\mathsf{Maybe}(\kappa)$ types are kept in \emph{both} branches: the true branch refines them to $\mathsf{Definite}(\kappa)$, and the false branch refines them to $\mathsf{None}$.
This ensures predicate-soundness for both branches after normalization.
Define defaults:
\begin{itemize}
\item $d_{\texttt{"string"}}=\mathsf{Some}(\texttt{string})$, $d_{\texttt{"number"}}=\mathsf{Some}(\texttt{number})$, $d_{\texttt{"bigint"}}=\mathsf{Some}(\texttt{bigint})$, $d_{\texttt{"boolean"}}=\mathsf{Some}(\texttt{boolean})$.
\item Let $F_{\emptyset}$ be the empty object field specification (no required/optional fields, no index signature, no \texttt{new}/\texttt{call} signature), and set $d_{\texttt{"object"}}=\mathsf{Some}(\texttt{object}(F_{\emptyset}))$.
\item $d_{\texttt{"function"}}=\mathsf{None}$ and $d_{\texttt{"undefined"}}=\mathsf{None}$.
\end{itemize}
\paragraph{Symbol.}
For $k=\texttt{"symbol"}$, SKJS performs no refinement, so we omit $f_k,g_k,d_k$.
\end{definition}

\paragraph{\texttt{typeof}.}
\begin{mathpar}
\inferrule*[right=R-TypeofT]{ }{\RefineT{\Env}{(\texttt{typeof}\ x === k)}{\mathsf{upd}(\Env,x,\,t \mapsto \mathsf{def}(t' \mapsto \refineop^{f_k,g_k}(t'),\,d_k)(t))}}
\and
\inferrule*[right=R-TypeofF]{ }{\RefineF{\Env}{(\texttt{typeof}\ x === k)}{\mathsf{upd}(\Env,x,\,t \mapsto \refineop^{f^{\neg}_k,g^{\neg}_k}(t))}}
\end{mathpar}
where $f_k,g_k,d_k$ are as in Definition~\ref{def:typeof-filters-defaults}, and $(f^{\neg}_k, g^{\neg}_k) = (\neg f_k, g_{\mathrm{id}})$ for most $k$, except for $k=\texttt{"function"}$ where $(f^{\neg}_{\texttt{"function"}}, g^{\neg}_{\texttt{"function"}}) = (f_{\neg\texttt{"function"}}, g_{\neg\texttt{"function"}})$ as defined in Definition~\ref{def:typeof-filters-defaults}.

\paragraph{\texttt{typeof} \texttt{"symbol"}.}
The implementation performs no refinement for \texttt{typeof x === "symbol"}; both branches leave the environment unchanged.

\paragraph{\texttt{in}.}
\begin{mathpar}
\inferrule*[right=R-InT]{ }{\RefineT{\Env}{(\texttt{"}p\texttt{" in }x)}{\mathsf{upd}(\Env,x,\,t \mapsto \mathsf{def}(t' \mapsto \refineop^{f^{\mathsf{true}}_p,g^{\mathsf{true}}_p}(t'),\,d^{\mathsf{in}}_p)(t))}}
\and
\inferrule*[right=R-InF]{ }{\RefineF{\Env}{(\texttt{"}p\texttt{" in }x)}{\mathsf{upd}(\Env,x,\,t \mapsto \refineop^{f^{\mathsf{false}}_p,g^{\mathsf{false}}_p}(t))}}
\end{mathpar}
where $d^{\mathsf{in}}_p=\mathsf{Some}(\texttt{object}(F_p))$ is a default object type requiring property $p$ with top value type (Definition~\ref{def:object-semantics}).
This corresponds to the implementation's behavior when the left-hand side is a single string literal.

\paragraph{Algebraic tag equality.}
\begin{mathpar}
\inferrule*[right=R-AlgTagT]{ }{\RefineT{\Env}{(x.\texttt{\_tag}===\ell)}{\mathsf{upd}(\Env,x,\,t \mapsto \refineop^{f^{\mathsf{alg}}_\ell,g^{\mathsf{alg,true}}_\ell}(t))}}
\and
\inferrule*[right=R-AlgTagF]{ }{\RefineF{\Env}{(x.\texttt{\_tag}===\ell)}{\mathsf{upd}(\Env,x,\,t \mapsto \refineop^{f^{\mathsf{alg}}_\ell,g^{\mathsf{alg,false}}_\ell}(t))}}
\end{mathpar}
where $\ell \in \String$ is a string literal.

\paragraph{\texttt{instanceof}.}
\begin{mathpar}
\inferrule*[right=R-InstT]{ }{\RefineT{\Env}{(x\ \texttt{instanceof}\ C)}{\mathsf{upd}(\Env,x,\,t \mapsto T_{\mathsf{inst}}(C))}}
\and
\inferrule*[right=R-InstF]{ }{\RefineF{\Env}{(x\ \texttt{instanceof}\ C)}{\Env}}
\end{mathpar}
where $T_{\mathsf{inst}}(C)$ is the instance type produced by the implementation (Definition~\ref{def:instanceof-semantics}).

\begin{mathpar}
\inferrule*[right=R-InstArrayT]{ }{\RefineT{\Env}{(x\ \texttt{instanceof}\ \texttt{Array})}{\mathsf{upd}(\Env,x,\,t \mapsto T_{\mathsf{array}})}}
\end{mathpar}
where $T_{\mathsf{array}}$ is the implementation's chosen array instance type (e.g. \texttt{Array}$\langle\top\rangle$).

\paragraph{Boolean connectives.}
The implementation composes refinements sequentially. For example:
\begin{mathpar}
\inferrule*[right=R-AndT]
{\RefineT{\Env}{e_1}{\Env_1} \\ \RefineT{\Env_1}{e_2}{\Env_2}}
{\RefineT{\Env}{(e_1 \ \&\&\ e_2)}{\Env_2}}
\and
\inferrule*[right=R-OrF]
{\RefineF{\Env}{e_1}{\Env_1} \\ \RefineF{\Env_1}{e_2}{\Env_2}}
{\RefineF{\Env}{(e_1 \ ||\ e_2)}{\Env_2}}
\end{mathpar}

\begin{lemma}[Predicate-Sound Transformers Yield Branch-Sound Rules]
\label{lem:predicate-to-branch-sound}
Let $R : \TType \to \TType$ be predicate-sound for predicate $P \subseteq \Val$.
Suppose the inference rule for expression $e$ updates the environment via $\mathsf{upd}(\Env, x, R)$.
If $\sigma \models_{\mathsf{true}} e$ implies $\sigma(x) \in P$, then the rule is branch-sound for the true branch.
Analogously for the false branch with $\sigma \models_{\mathsf{false}} e$.
\end{lemma}

\begin{proof}
Suppose $\sigma \models \Env$ and $\sigma \models_{\mathsf{true}} e$.
Let $\Env' = \mathsf{upd}(\Env, x, R)$.
For any local $y \neq x$, $\Env'(y) = \Env(y)$ and $\sigma(y) \in \sem{\Env(y)} = \sem{\Env'(y)}$.
For $x$: by hypothesis $\sigma(x) \in P$, and by $\sigma \models \Env$ we have $\sigma(x) \in \sem{\Env(x)}$.
Thus $\sigma(x) \in \sem{\Env(x)} \cap P \subseteq \sem{R(\Env(x))} = \sem{\Env'(x)}$ by predicate-soundness of $R$.
Hence $\sigma \models \Env'$.
\end{proof}

\begin{lemma}[Connective Rules are Branch-Sound]
\label{lem:connectives-branch-sound}
Assume that for every derivation $\RefineT{\Env}{e_1}{\Env_1}$, the rule is branch-sound for the true branch in the sense of Definition~\ref{def:refinement}.
Assume likewise that for every derivation $\RefineT{\Env_1}{e_2}{\Env_2}$, the rule is branch-sound for the true branch.
Then the connective rule \textbf{R-AndT} is branch-sound for the true branch.

Dually, assume that for every derivation $\RefineF{\Env}{e_1}{\Env_1}$ and $\RefineF{\Env_1}{e_2}{\Env_2}$, the rules are branch-sound for the false branch.
Then the connective rule \textbf{R-OrF} is branch-sound for the false branch.
\end{lemma}

\begin{proof}
\textbf{R-AndT.}
Suppose $\sigma \models \Env$ and $\sigma \models_{\mathsf{true}} (e_1 \ \&\&\ e_2)$.
By Definition~\ref{def:refine-eval}, we have $\sigma \models_{\mathsf{true}} e_1$ and $\sigma \models_{\mathsf{true}} e_2$.
Consider a derivation of $\RefineT{\Env}{(e_1 \ \&\&\ e_2)}{\Env_2}$ via \textbf{R-AndT}; it contains premises
$\RefineT{\Env}{e_1}{\Env_1}$ and $\RefineT{\Env_1}{e_2}{\Env_2}$.
By branch-soundness of the first premise and $\sigma \models_{\mathsf{true}} e_1$, from $\sigma \models \Env$ we obtain $\sigma \models \Env_1$.
Then by branch-soundness of the second premise and $\sigma \models_{\mathsf{true}} e_2$, we obtain $\sigma \models \Env_2$.
This is exactly branch-soundness of \textbf{R-AndT}.

\textbf{R-OrF.}
Suppose $\sigma \models \Env$ and $\sigma \models_{\mathsf{false}} (e_1 \ ||\ e_2)$.
By Definition~\ref{def:refine-eval}, we have $\sigma \models_{\mathsf{false}} e_1$ and $\sigma \models_{\mathsf{false}} e_2$.
For a derivation of $\RefineF{\Env}{(e_1 \ ||\ e_2)}{\Env_2}$ via \textbf{R-OrF}, we similarly obtain $\sigma \models \Env_1$ from the first premise and then $\sigma \models \Env_2$ from the second premise.
\end{proof}

\paragraph{Soundness connection.}
Each of the case-study lemmas in the following subsections establishes that the filter/map pair $(f,g)$ is sound for the corresponding runtime predicate $P$, 
which by Lemma~\ref{lem:filter-map-predicate-sound} implies predicate-soundness of $\refineop^{f,g}$,
and by Lemma~\ref{lem:predicate-to-branch-sound} implies branch-soundness of the inference rule.

\subsection{Soundness Proofs}
\label{sec:refine-proofs}

The following subsections prove soundness for each refinement category.
Each proof shows that the filter/map pair $(f,g)$ is sound for the runtime predicate $P$,
then applies Lemmas~\ref{lem:filter-map-predicate-sound} and~\ref{lem:predicate-to-branch-sound}.

\subsubsection{Property Check (\texttt{in})}

The \texttt{"p" in x} refinement adds a required field on the true branch, or removes the field constraint on the false branch.

\begin{definition}[\texttt{in} Predicates]
\label{def:in-predicate}
Fix a property name $p \in \String$.
Define:
\[
P_{\mathrm{in}}^{\mathsf{true}}(p) \triangleq \{\, o \in \Val \mid o \in \ObjVal \land \mathsf{has}(o,p)\,\}
\qquad
P_{\mathrm{in}}^{\mathsf{false}}(p) \triangleq \Val \setminus P_{\mathrm{in}}^{\mathsf{true}}(p).
\]
\end{definition}

\begin{definition}[\texttt{in} Filter/Map Pair]
\label{def:in-filter-map}
Fix $p \in \String$ and let $\top_{\TType}$ be any type with $\sem{\top_{\TType}}=\Val$ (e.g. the unique top clause).
Let $F_p$ be the object field specification with $F_{\mathrm{req}}$ mapping $p \mapsto \top_{\TType}$, $F_{\mathrm{opt}}$ empty, $F_{\mathrm{idx}}=\mathsf{None}$, and $F_{\mathrm{new}}=F_{\mathrm{call}}=\mathsf{None}$.
Define a mapper $g^{\mathsf{true}}_p : \SType \to \SType$ as follows:
\begin{itemize}
\item If $s$ is an object type $s=\texttt{object}(F)$ and $F_{\mathrm{idx}}=\mathsf{None}$ and $p \notin \mathrm{dom}(F_{\mathrm{req}})$, then $g^{\mathsf{true}}_p(s)$ is $\texttt{object}(F')$ where $F'$ is $F$ except that $F'_{\mathrm{req}}$ additionally maps $p \mapsto \top_{\TType}$.
\item Otherwise, $g^{\mathsf{true}}_p(s)=s$.
\end{itemize}
Let $f^{\mathsf{true}}_p : \SType \to \{\mathsf{true},\mathsf{false}\}$ be the filter defined by:
\[
f^{\mathsf{true}}_p(s) = 
\begin{cases}
\mathsf{true} & \text{if } \tagfn(s) \in \{7,8,9,10\} \text{ (tuple, promise, object, or algebraic)} \\
\mathsf{false} & \text{otherwise (primitives)}
\end{cases}
\]
\paragraph{Note.}
The implementation conservatively retains tuples, promises, and algebraic types (tags 7, 8, 10) in addition to objects; these cases are marked \texttt{TODO} in the implementation.
Only object-tagged values (tag 9) can actually satisfy the \texttt{in} predicate at runtime, so the filter is sound but imprecise for tags 7, 8, and 10.

Define a mapper $g^{\mathsf{false}}_p : \SType \to \SType$ by:
\begin{itemize}
\item If $s=\texttt{object}(F)$ then $g^{\mathsf{false}}_p(s)=\texttt{object}(F\setminus p)$ where $F\setminus p$ removes $p$ from both $F_{\mathrm{req}}$ and $F_{\mathrm{opt}}$ (and leaves $F_{\mathrm{idx}},F_{\mathrm{new}},F_{\mathrm{call}}$ unchanged).
\item Otherwise $g^{\mathsf{false}}_p(s)=s$.
\end{itemize}
Let $f^{\mathsf{false}}_p(s)=\mathsf{true}$ for all $s$ (the implementation does not attempt to exclude non-objects in the false branch).
\paragraph{Default.}
Define $d^{\mathsf{in}}_p \triangleq \mathsf{Some}(\texttt{object}(F_p))$.
\end{definition}

\begin{lemma}[\texttt{in}-True Branch (with Default) is Predicate-Sound]
\label{lem:in-true-predicate-sound}
For every $p \in \String$, the transformer $t \mapsto \mathsf{def}(t' \mapsto \refineop^{f^{\mathsf{true}}_p,g^{\mathsf{true}}_p}(t'), d^{\mathsf{in}}_p)(t)$
is predicate-sound for $P_{\mathrm{in}}^{\mathsf{true}}(p)$.
\end{lemma}

\begin{proof}
Let $R_p(t)\triangleq \refineop^{f^{\mathsf{true}}_p,g^{\mathsf{true}}_p}(t)$.
We first show that $(f^{\mathsf{true}}_p,g^{\mathsf{true}}_p)$ is sound for $P_{\mathrm{in}}^{\mathsf{true}}(p)$ (Definition~\ref{def:filter-map-refinement}), and conclude that $R_p$ is predicate-sound for $P_{\mathrm{in}}^{\mathsf{true}}(p)$ by Lemma~\ref{lem:filter-map-predicate-sound}.
Let $s \in \SType$ and $v \in \sem{s}$ with $v \in P_{\mathrm{in}}^{\mathsf{true}}(p)$.
Then $v \in \ObjVal$, so $\rtag(v)=9$.
By A0, $\rtag(v)=\tagfn(s)$, hence $\tagfn(s)=9$ and therefore $s$ must be an object type of the form $\texttt{object}(F)$.
By definition of $P_{\mathrm{in}}^{\mathsf{true}}(p)$, $\mathsf{has}(v,p)$.
The filter keeps object types, so $f^{\mathsf{true}}_p(s)=\mathsf{true}$.
If $g^{\mathsf{true}}_p$ adds the required field $p \mapsto \top_{\TType}$, then $v$ satisfies the added requirement because it already has $p$ and $\mathsf{get}(v,p) \in \sem{\top_{\TType}}=\Val$.
Otherwise $g^{\mathsf{true}}_p(s)=s$ and membership is immediate.
Thus $v \in \sem{g^{\mathsf{true}}_p(s)}$ and soundness holds.

Finally, by Definition~\ref{def:object-semantics}, every $v \in P_{\mathrm{in}}^{\mathsf{true}}(p)$ is in $\sem{\texttt{object}(F_p)}=\sem{\mathsf{single}(\texttt{object}(F_p))}$, so
$P_{\mathrm{in}}^{\mathsf{true}}(p) \subseteq \sem{\mathsf{single}(\texttt{object}(F_p))}$.
By Lemma~\ref{lem:defaulting-preserves-predicate-sound}, $\mathsf{def}(R_p,d^{\mathsf{in}}_p)$ is predicate-sound for $P_{\mathrm{in}}^{\mathsf{true}}(p)$.
\end{proof}

\begin{lemma}[\texttt{in}-False Branch is Predicate-Sound]
\label{lem:in-false-predicate-sound}
For every $p \in \String$, the refinement $\refineop^{f^{\mathsf{false}}_p,g^{\mathsf{false}}_p}$ is predicate-sound for $P_{\mathrm{in}}^{\mathsf{false}}(p)$.
\end{lemma}

\begin{proof}
We show $(f^{\mathsf{false}}_p,g^{\mathsf{false}}_p)$ is sound for $P_{\mathrm{in}}^{\mathsf{false}}(p)$ (Definition~\ref{def:filter-map-refinement}), and conclude by Lemma~\ref{lem:filter-map-predicate-sound}.
Let $s \in \SType$ and $v \in \sem{s}$ with $v \in P_{\mathrm{in}}^{\mathsf{false}}(p)$.
We have $f^{\mathsf{false}}_p(s)=\mathsf{true}$ by definition.
If $s$ is not an object type, then $g^{\mathsf{false}}_p(s)=s$ so $v \in \sem{g^{\mathsf{false}}_p(s)}$.
If $s=\texttt{object}(F)$, then $g^{\mathsf{false}}_p(s)=\texttt{object}(F\setminus p)$ which removes $p$ from field constraints.
By Definition~\ref{def:object-semantics}, the semantics only constrains listed fields, so removing a field enlarges (or preserves) the inhabitant set: $\sem{\texttt{object}(F)} \subseteq \sem{\texttt{object}(F\setminus p)}$.
Hence $v \in \sem{g^{\mathsf{false}}_p(s)}$.
\end{proof}

\subsubsection{Nullish Checks}

\begin{definition}[Nullish Predicates]
\label{def:nullish-predicates}
Define the following predicates on runtime values:
\[
P_{\mathsf{null}} \triangleq \{ v \in \Val \mid \rtag(v)=2 \}
\qquad
P_{\mathsf{undef}} \triangleq \{ v \in \Val \mid \rtag(v)\in\{0,1\} \}.
\]
We also write $\neg P$ for $\Val \setminus P$.
\end{definition}

\begin{lemma}[Nullish Refinement is Predicate-Sound]
\label{lem:nullish-predicate-sound}
The refinements for null and undefined (Definition~\ref{def:nullish-filters-mappers}) are predicate-sound for their respective predicates.
\end{lemma}

\begin{proof}
By A0: if $v \in P_{\mathsf{null}}$ then $\rtag(v)=2$, so $\tagfn(s)=2$ and $f_{\mathsf{null}}(s)=\mathsf{true}$.
Apply Lemma~\ref{lem:filter-map-predicate-sound}. The undefined and negated cases are analogous.
\end{proof}

\subsubsection{\texttt{typeof} Checks}

Predicates are in Definition~\ref{def:typeof-predicates}; filters/defaults in Definition~\ref{def:typeof-filters-defaults}.

\begin{lemma}[\texttt{typeof} Refinement is Predicate-Sound]
\label{lem:typeof-refine-predicate-sound}
For each $k$ in Definition~\ref{def:typeof-predicates}, the transformer $t \mapsto \refineop^{f_k,g_k}(t)$ is predicate-sound for $P_{\mathsf{typeof}=k}$,
and the false-branch transformer $t \mapsto \refineop^{f^{\neg}_k,g^{\neg}_k}(t)$ is predicate-sound for $\neg P_{\mathsf{typeof}=k}$,
where $(f^{\neg}_k, g^{\neg}_k) = (\neg f_k, g_{\mathrm{id}})$ for most $k$, except for $k=\texttt{"function"}$ where $(f^{\neg}_{\texttt{"function"}}, g^{\neg}_{\texttt{"function"}}) = (f_{\neg\texttt{"function"}}, g_{\neg\texttt{"function"}})$ as in Definition~\ref{def:typeof-filters-defaults}.
\paragraph{Defaults.}
If $d_k=\mathsf{Some}(s_k)$ satisfies $P_{\mathsf{typeof}=k} \subseteq \sem{\mathsf{single}(s_k)}$, then the defaulted transformer
\[
t \mapsto \mathsf{def}(t' \mapsto \refineop^{f_k,g_k}(t'), d_k)(t)
\]
is predicate-sound for $P_{\mathsf{typeof}=k}$.
\end{lemma}

\begin{proof}
For tag-based cases ($k \in \{\texttt{"string"}, \texttt{"number"}, \ldots\}$): by A0, $v \in P_{\mathsf{typeof}=k}$ implies $\tagfn(s)$ matches the predicate's tag-set, so $f_k(s)=\mathsf{true}$.
Since $g_k = g_{\mathrm{id}}$, apply Lemma~\ref{lem:filter-map-predicate-sound}.

For $k=\texttt{"function"}$: the key is that $\mathsf{callable}(v)=\mathsf{true}$ forces $F_{\mathrm{call}} \neq \mathsf{None}$, so the mapper upgrading $\mathsf{Maybe}(\kappa) \to \mathsf{Definite}(\kappa)$ preserves $v$.

For the false branch of \texttt{"function"}: $\mathsf{callable}(v)=\mathsf{false}$ implies $F_{\mathrm{call}} \in \{\mathsf{None}, \mathsf{Maybe}(\kappa)\}$, and the mapper downgrading to $\mathsf{None}$ preserves $v$.

Defaults follow from Lemma~\ref{lem:defaulting-preserves-predicate-sound}.
\end{proof}

\subsubsection{Discriminant (\texttt{\_tag}) Checks}

Refines algebraic types by filtering the constructor tag set.

\begin{definition}[Algebraic-Tag Predicate]
\label{def:alg-tag-predicate}
Fix $\ell \in \String$ (a tag label) and define:
\[
P_{\mathsf{algTag}=\ell} \triangleq \{ v \in \Val \mid v \in \AlgVal \land \mathsf{atag}(v)=\ell \}.
\]
\end{definition}

\begin{definition}[Algebraic Tag Filter/Map Pairs]
\label{def:alg-tag-filter-map}
Fix $\ell \in \String$.
Let $f^{\mathsf{alg}}_\ell(s)=\mathsf{true}$ for all $s$ (the implementation does not drop non-algebraic alternatives for this refinement).
Define mappers:
\begin{itemize}
\item $g^{\mathsf{alg,true}}_\ell(\texttt{algebraic}(T)) \triangleq \texttt{algebraic}(T \cap \{\ell\})$, and $g^{\mathsf{alg,true}}_\ell(s)\triangleq s$ for all other $s$.
\item $g^{\mathsf{alg,false}}_\ell(\texttt{algebraic}(T)) \triangleq \texttt{algebraic}(T \setminus \{\ell\})$, and $g^{\mathsf{alg,false}}_\ell(s)\triangleq s$ for all other $s$.
\end{itemize}
\end{definition}

\begin{lemma}[Algebraic Tag Equality is Predicate-Sound]
\label{lem:alg-tag-predicate-sound}
For every $\ell \in \String$:
\begin{align*}
&\refineop^{f^{\mathsf{alg}}_\ell,g^{\mathsf{alg,true}}_\ell} \text{ is predicate-sound for } P_{\mathsf{algTag}=\ell}, \\
&\refineop^{f^{\mathsf{alg}}_\ell,g^{\mathsf{alg,false}}_\ell} \text{ is predicate-sound for } \neg P_{\mathsf{algTag}=\ell}.
\end{align*}
\end{lemma}

\begin{proof}
True branch: If $v \in P_{\mathsf{algTag}=\ell}$ then $\mathsf{atag}(v)=\ell$, so $\ell \in T$ for $s=\texttt{algebraic}(T)$, and the mapper restricting to $\{ell\}$ preserves $v$.

False branch: If $\mathsf{atag}(v) \neq \ell$, then $\mathsf{atag}(v) \in T \setminus \{\ell\}$, so the mapper removing $\ell$ preserves $v$. Non-algebraic types pass through unchanged.
\end{proof}

\subsubsection{\texttt{instanceof} Checks}

Refines the true branch by overwriting with the instance type $T_{\mathsf{inst}}(C)$.

\begin{definition}[\texttt{instanceof} Semantics and Implementation-Modeled Refinement]
\label{def:instanceof-semantics}
Assume a set of runtime class names $\mathsf{Class}$ and a predicate family
\[
P_{\mathsf{inst}} : \mathsf{Class} \to \Pred
\]
capturing the runtime truth of \texttt{instanceof} for each class.
We assume the type system's instance forms in $\TType$ include a corresponding instance type $T_{\mathsf{inst}}(C)$ (with implementation-chosen type arguments), and a distinguished array instance type $T_{\mathsf{array}}$.

\paragraph{Axiom (\texttt{instanceof} inclusion).}
For each class $C$ we assume:
\[
v \in P_{\mathsf{inst}}(C) \implies v \in \sem{T_{\mathsf{inst}}(C)}.
\]
\paragraph{Array special case.}
We assume $P_{\mathsf{inst}}(\texttt{Array}) \subseteq \sem{T_{\mathsf{array}}}$.
\end{definition}

\begin{lemma}[\texttt{instanceof}-True Overwrite is Predicate-Sound]
\label{lem:instanceof-predicate-sound}
For each class $C$, the constant transformer $R_C(t)\triangleq T_{\mathsf{inst}}(C)$ is predicate-sound for $P_{\mathsf{inst}}(C)$.
Similarly, the constant transformer $R_{\mathsf{array}}(t)\triangleq T_{\mathsf{array}}$ is predicate-sound for $P_{\mathsf{inst}}(\texttt{Array})$.
\end{lemma}

\begin{proof}
Fix $C$ and let $t \in \TType$.
If $v \in \sem{t}\cap P_{\mathsf{inst}}(C)$, then $v \in P_{\mathsf{inst}}(C)$.
By the axiom in Definition~\ref{def:instanceof-semantics}, $v \in \sem{T_{\mathsf{inst}}(C)} = \sem{R_C(t)}$.
Thus $\sem{t}\cap P_{\mathsf{inst}}(C) \subseteq \sem{R_C(t)}$.
The array case is identical using $P_{\mathsf{inst}}(\texttt{Array}) \subseteq \sem{T_{\mathsf{array}}}$.
\end{proof}

\section{Type Difference (Diff)}

\begin{definition}[Diff Soundness Properties]
\label{def:diff-soundness}
Let $X$ be a set equipped with a semantic interpretation $\sem{\cdot} : X \to \mathcal{P}(\Val)$.
An operation $\diff : X \times X \to X$ is \emph{sound} (with respect to $\sem{\cdot}$) if for all $x_1,x_2 \in X$, letting $x_3 = \diff(x_1,x_2)$, the following properties hold:
\begin{align}
\text{(P1)} \quad & \sem{x_3} \subseteq \sem{x_1} \tag{under-approximation} \\
\text{(P2)} \quad & \sem{x_3} \cup \sem{x_2} \supseteq \sem{x_1} \tag{coverage}
\end{align}
In this document we instantiate $X$ with $\STypes$, $\Clause$, $\Clauses$, and $\TType$.
\paragraph{Reader's guide.}
These two properties imply that $\sem{x_3}$ is always a refinement of $\sem{x_1}$, and that it still covers all of $\sem{x_1}$ once $\sem{x_2}$ is added back.
Equivalently (Lemma~\ref{lem:diff-equiv-char}), $\sem{x_3}$ is an \emph{over-approximation} of the true set difference $\sem{x_1}\setminus\sem{x_2}$ while staying within $\sem{x_1}$.
\end{definition}

\begin{lemma}[Equivalent Characterization]
\label{lem:diff-equiv-char}
For all $x_1,x_2,x_3 \in X$, (P1) $\land$ (P2) holds if and only if
\[
\sem{x_1} \setminus \sem{x_2} \subseteq \sem{x_3} \subseteq \sem{x_1}.
\]
\end{lemma}

\begin{proof}
($\Rightarrow$) Let $v \in \sem{x_1} \setminus \sem{x_2}$. By (P2),
$v \in \sem{x_3} \cup \sem{x_2}$. Since $v \notin \sem{x_2}$, we have $v \in \sem{x_3}$.
Thus $\sem{x_1} \setminus \sem{x_2} \subseteq \sem{x_3}$. Combined with (P1):
$\sem{x_1} \setminus \sem{x_2} \subseteq \sem{x_3} \subseteq \sem{x_1}$.

($\Leftarrow$) The right-hand inclusion $\sem{x_3} \subseteq \sem{x_1}$ is exactly (P1).
For (P2),
\[
\sem{x_3} \cup \sem{x_2} \supseteq (\sem{x_1} \setminus \sem{x_2}) \cup \sem{x_2} \supseteq \sem{x_1}.
\]
\end{proof}

\begin{definition}[Completeness]
A diff operation is \emph{complete} if:
\[
\sem{\diff(t_1, t_2)} = \sem{t_1} \setminus \sem{t_2}
\]
By the equivalent characterization, (P1) $\land$ (P2) implies 
$\sem{t_1} \setminus \sem{t_2} \subseteq \sem{\diff(t_1, t_2)}$.
Completeness additionally requires the reverse: $\sem{\diff(t_1, t_2)} \subseteq \sem{t_1} \setminus \sem{t_2}$.
\end{definition}

\subsection{STypes Diff}

\begin{definition}[STypes Diff]
\label{def:stypes-diff}
\[
\diff_{\STypes}(L, R) \triangleq \{\, s \in L \mid \tagfn(s) \notin \tags(R)\,\}.
\]
\paragraph{Intuition.}
Since an $\STypes$ value stores at most one representative per runtime tag, this diff only removes whole tags:
it preserves exactly the tags present on the left but absent on the right.
\end{definition}

\begin{lemma}[STypes Diff Satisfies (P1)]
\label{lem:stypes-diff-p1}
$\sem{\diff_{\STypes}(L, R)} \subseteq \sem{L}$.
\end{lemma}

\begin{proof}
\(\diff_{\STypes}(L,R) \subseteq L\) by definition, so:
\[
\sem{\diff_{\STypes}(L,R)}
= \bigcup_{s \in \diff_{\STypes}(L,R)} \sem{s}
\subseteq \bigcup_{s \in L} \sem{s}
= \sem{L}.
\]
\end{proof}

\begin{definition}[Same-Tag Coverage Condition]
\label{def:stc}
For $L, R \in \STypes$, define:
\[
\mathsf{STC}(L, R) \iff \forall k \in \tags(L) \cap \tags(R).\; \sem{L[k]} \subseteq \sem{R[k]}.
\]
\end{definition}

\begin{lemma}[STypes Diff Satisfies (P2) under STC]
\label{lem:stypes-diff-p2-under-stc}
If $\mathsf{STC}(L, R)$ holds, then $\sem{\diff_{\STypes}(L, R)} \cup \sem{R} \supseteq \sem{L}$.
\end{lemma}

\begin{proof}
Let $x \in \sem{L}$. Then $x \in \sem{L[k]}$ for some $k \in \tags(L)$.
\begin{itemize}
\item Case $k \notin \tags(R)$: Let \(s=L[k]\). Then \(\tagfn(s)=k\) and \(s \in \diff_{\STypes}(L,R)\), so \(x \in \sem{s} \subseteq \sem{\diff_{\STypes}(L,R)}\).
\item Case $k \in \tags(R)$: By $\mathsf{STC}$, \(\sem{L[k]} \subseteq \sem{R[k]} \subseteq \sem{R}\), so \(x \in \sem{R}\).
\end{itemize}
\end{proof}

\begin{corollary}[STypes Diff under STC]
\label{cor:stypes-diff-under-stc}
If $\mathsf{STC}(L, R)$ holds, then (P2) holds.
In particular, if $L[k] = R[k]$ for all $k \in \tags(L) \cap \tags(R)$, then (P2) holds.
\end{corollary}

\begin{lemma}[STypes Diff Completeness]
\label{lem:stypes-diff-complete-iff-stc}
$\diff_{\STypes}$ is complete (i.e., $\sem{\diff_{\STypes}(L, R)} = \sem{L} \setminus \sem{R}$) 
if and only if $\mathsf{STC}(L, R)$.
\end{lemma}

\begin{proof}
($\Rightarrow$) Suppose completeness holds and let $k \in \tags(L) \cap \tags(R)$.
Let \(s_L=L[k]\) and \(s_R=R[k]\).
Since \(k \in \tags(R)\), we have \(s_L \notin \diff_{\STypes}(L,R)\), so by completeness \(\sem{s_L} \subseteq \sem{R}\).
By A1 and the distinct-tag invariant for \(\STypes\), \(\sem{s_L}\) can only overlap with the unique same-tag alternative \(\sem{s_R}\), hence \(\sem{s_L} \subseteq \sem{s_R}\), i.e. \(\sem{L[k]} \subseteq \sem{R[k]}\).

($\Leftarrow$) We show $\sem{\diff_{\STypes}(L, R)} = \sem{L} \setminus \sem{R}$.

($\subseteq$) Let $x \in \sem{\diff_{\STypes}(L, R)}$. Then $x \in \sem{s}$ for some $s \in \diff_{\STypes}(L,R)$.
Let \(k=\tagfn(s)\). Then \(k \notin \tags(R)\), so by A1 no element of \(R\) has tag \(k\), hence \(x \notin \sem{R}\).
Also \(s \in L\), so \(x \in \sem{L}\). Therefore \(x \in \sem{L} \setminus \sem{R}\).

($\supseteq$) Let $x \in \sem{L} \setminus \sem{R}$. Then \(x \in \sem{L[k]}\) for some \(k \in \tags(L)\).
If \(k \in \tags(R)\), then by hypothesis \(\sem{L[k]} \subseteq \sem{R[k]} \subseteq \sem{R}\), contradicting \(x \notin \sem{R}\).
So \(k \notin \tags(R)\), hence \(L[k] \in \diff_{\STypes}(L,R)\) and \(x \in \sem{\diff_{\STypes}(L, R)}\).
\end{proof}

\begin{corollary}[When STypes Diff is Incomplete]
$\diff_{\STypes}(L, R)$ is incomplete when there exists $k \in \tags(L) \cap \tags(R)$ with $\sem{L[k]} \not\subseteq \sem{R[k]}$.
Example: $\diff_{\STypes}(\{\texttt{number}\}, \{\texttt{1}\}) = \emptyset$, 
but $\sem{\texttt{number}} \setminus \sem{\texttt{1}} \neq \emptyset$.
\end{corollary}

\subsection{TInsts Diff}

\begin{definition}[TInsts Diff]
\label{def:tinsts-diff}
\[
\diff_{\TInsts}(L, R) = L \setminus R = \{ i \in L \mid i \notin R \}
\]
\paragraph{Note.}
In the implementation, $\TInsts$ is represented as a collection of instance AST nodes and $\diff_{\TInsts}$ removes elements based on \emph{structural equality} of instances.
In this paper, that corresponds exactly to the usual mathematical set difference on the set $\TInst$ (i.e. membership/equality is the syntactic identity of instances, not semantic overlap).
\end{definition}

\begin{lemma}[TInsts Diff Satisfies (P1) and (P2)]
\label{lem:tinsts-diff-sound}
$\sem{\diff_{\TInsts}(L, R)} \subseteq \sem{L}$ and 
$\sem{\diff_{\TInsts}(L, R)} \cup \sem{R} \supseteq \sem{L}$.
\end{lemma}

\begin{proof}
(P1): $\diff_{\TInsts}(L, R) \subseteq L$, so $\sem{\diff_{\TInsts}(L, R)} \subseteq \sem{L}$.

(P2): Let $x \in \sem{L}$. Then $x \in \sem{i}$ for some $i \in L$.
\begin{itemize}
\item If $i \in R$: Then $x \in \sem{i} \subseteq \sem{R}$.
\item If $i \notin R$: Then $i \in \diff_{\TInsts}(L, R)$, so $x \in \sem{\diff_{\TInsts}(L, R)}$.
\end{itemize}
\end{proof}

\begin{lemma}[TInsts Diff Completeness]
\label{lem:tinsts-diff-complete}
$\diff_{\TInsts}$ is complete if and only if for all $i \in L \setminus R$ and $j \in R$:
$\sem{i} \cap \sem{j} = \emptyset$.
\end{lemma}

\begin{proof}
($\Rightarrow$) Suppose completeness and let $i \in L \setminus R$ and $j \in R$.
If $x \in \sem{i} \cap \sem{j}$, then $i \in \diff_{\TInsts}(L, R)$, so $x \in \sem{\diff_{\TInsts}(L, R)} = \sem{L} \setminus \sem{R}$.
But $x \in \sem{j} \subseteq \sem{R}$, contradiction.

($\Leftarrow$) We show $\sem{\diff_{\TInsts}(L, R)} = \sem{L} \setminus \sem{R}$.

($\subseteq$) Let $x \in \sem{\diff_{\TInsts}(L, R)}$. Then $x \in \sem{i}$ for some $i \in L \setminus R$.
For any $j \in R$, $\sem{i} \cap \sem{j} = \emptyset$ by hypothesis.
Thus $x \notin \sem{R}$, so $x \in \sem{L} \setminus \sem{R}$.

($\supseteq$) Let $x \in \sem{L} \setminus \sem{R}$. Then $x \in \sem{i}$ for some $i \in L$.
If $i \in R$, then $x \in \sem{R}$, contradiction. So $i \in L \setminus R = \diff_{\TInsts}(L, R)$.
\end{proof}

\subsection{Clause Diff}

\begin{definition}[Clause Diff]
\label{def:clause-diff}
For clauses $c_1 = \clause{V_1}{S_1}$ and $c_2 = \clause{V_2}{S_2}$:
\[
\diff_{\Clause}(c_1, c_2) = 
\begin{cases}
\bot & \text{if } S_2 = \top \\
c_1 & \text{if } S_1 = \top \\
\clause{V_1}{\diff_{\STypes}(S_1, S_2)} & \text{otherwise}
\end{cases}
\]
where $\bot$ (empty type) has $\sem{\bot} = \emptyset$.
\end{definition}

\begin{lemma}[Clause Diff Satisfies (P1)]
\label{lem:clause-diff-p1}
$\sem{\diff_{\Clause}(c_1, c_2)} \subseteq \sem{c_1}$.
\end{lemma}

\begin{proof}
By case analysis on $S_1, S_2$. The non-trivial case uses Lemma~\ref{lem:stypes-diff-p1}.
\end{proof}

\begin{lemma}[Clause Diff Satisfies (P2) Conditionally]
\label{lem:clause-diff-p2-conditional}
$\diff_{\Clause}$ satisfies (P2) when:
\begin{enumerate}
\item $S_2 = \top$: Always (since $\sem{c_2} = \Val \supseteq \sem{c_1}$).
\item $S_1 = \top$: Always (result is $c_1$).
\item Otherwise: When $\diff_{\STypes}(S_1, S_2)$ satisfies (P2).
\end{enumerate}
\end{lemma}

\subsection{Clauses Diff}

\begin{definition}[Clauses Diff]
\label{def:clauses-diff}
For $C_1, C_2 \in \Clauses$, let $c_{\emptyset} = C_2(\emptyset)$ if $\emptyset \in \mathrm{dom}(C_2)$, else undefined.
\[
\diff_{\Clauses}(C_1, C_2)(k) = 
\begin{cases}
\diff_{\Clause}(\diff_{\Clause}(C_1(k), C_2(k)), c_{\emptyset}) & \text{if } k \in \mathrm{dom}(C_2) \land c_{\emptyset} \text{ defined} \\
\diff_{\Clause}(C_1(k), C_2(k)) & \text{if } k \in \mathrm{dom}(C_2) \land c_{\emptyset} \text{ undefined} \\
\diff_{\Clause}(C_1(k), c_{\emptyset}) & \text{if } k \notin \mathrm{dom}(C_2) \land c_{\emptyset} \text{ defined} \\
C_1(k) & \text{if } k \notin \mathrm{dom}(C_2) \land c_{\emptyset} \text{ undefined}
\end{cases}
\]
for $k \in \mathrm{dom}(C_1)$. Clauses mapped to $\bot$ are dropped.
\paragraph{Note.}
Whenever $k \in \mathrm{dom}(C_2)$, the call $\diff_{\Clause}(C_1(k), C_2(k))$ compares clauses stored under the \emph{same key} $k$, so their variable-sets match by the $\Clauses$ well-formedness condition (keys agree with clause variables).
The only subtraction that may use a different key is via $c_{\emptyset}=C_2(\emptyset)$, which is deliberately applied to every $C_1(k)$; this corresponds to the implementation behavior and explains why the right-hand clause's variables are not used by $\diff_{\Clause}$.
\paragraph{Why the $\emptyset$-clause matters.}
The distinguished key $\emptyset$ corresponds to clauses that do not require any type parameters.
When present on the right, subtracting $c_{\emptyset}$ from every left clause models a ``global'' elimination that should apply regardless of which type-parameter constraints produced the left clause.
\end{definition}

\begin{lemma}[Clauses Diff Satisfies (P1)]
\label{lem:clauses-diff-p1}
$\sem{\diff_{\Clauses}(C_1, C_2)} \subseteq \sem{C_1}$.
\end{lemma}

\begin{proof}
Each output clause is derived from $C_1(k)$ via $\diff_{\Clause}$, which satisfies (P1).
\end{proof}

\subsection{TType Diff}

\begin{definition}[TType Diff]
\label{def:ttype-diff}
For $t_1 = (\insts_1, \clauses_1)$ and $t_2 = (\insts_2, \clauses_2)$:
\[
\diff_{\TType}(t_1, t_2) = (\diff_{\TInsts}(\insts_1, \insts_2), \diff_{\Clauses}(\clauses_1, \clauses_2))
\]
\end{definition}

\begin{theorem}[TType Diff Satisfies (P1)]
\label{thm:ttype-diff-p1}
$\sem{\diff_{\TType}(t_1, t_2)} \subseteq \sem{t_1}$.
\end{theorem}

\begin{proof}
\begin{align*}
\sem{\diff_{\TType}(t_1, t_2)} &= \sem{\diff_{\TInsts}(\insts_1, \insts_2)} \cup \sem{\diff_{\Clauses}(\clauses_1, \clauses_2)} \\
&\subseteq \sem{\insts_1} \cup \sem{\clauses_1} = \sem{t_1}
\end{align*}
by Lemmas~\ref{lem:tinsts-diff-sound} and~\ref{lem:clauses-diff-p1}.
\end{proof}

\begin{theorem}[TType Diff Satisfies (P2) Conditionally]
$\diff_{\TType}$ satisfies (P2) when all underlying $\diff_{\STypes}$ calls satisfy (P2).
In particular, (P2) holds when $\mathsf{STC}$ holds for all same-tag eliminations.
\end{theorem}

\begin{theorem}[TType Diff Completeness]
$\diff_{\TType}$ is complete when:
\begin{enumerate}
\item $\diff_{\TInsts}$ is complete (structurally distinct instances are semantically disjoint), and
\item $\diff_{\STypes}$ is complete for all clause pairs (same-tag implies $\sem{s} \subseteq \sem{s'}$).
\end{enumerate}
\end{theorem}

\begin{corollary}[Explicit Completeness Conditions for $\diff_{\TType}$]
Let $t_1=(\insts_1,\clauses_1)$ and $t_2=(\insts_2,\clauses_2)$.
Then $\diff_{\TType}(t_1,t_2)$ is complete (i.e. $\sem{\diff_{\TType}(t_1,t_2)}=\sem{t_1}\setminus\sem{t_2}$) under the following explicit hypotheses:
\begin{enumerate}
\item \textbf{Instance disjointness:}
  For all $i\in\insts_1$ and $j\in\insts_2$ with $i\neq j$, we have $\sem{i}\cap\sem{j}=\emptyset$.

\item \textbf{Top-clause alignment:}
  If $\clauses_1$ contains $\clause{\emptyset}{\top}$, then $\clauses_2$ also contains it.

\item \textbf{Same-tag coverage for clause subtractions:}
  For each $k \in \mathrm{dom}(\clauses_1)\cap \mathrm{dom}(\clauses_2)$ with $\clauses_i(k)=\clause{k}{S_i}$: $\mathsf{STC}(S_1,S_2)$.
  Similarly for the empty-key clause if present in $\clauses_2$.
\end{enumerate}
\end{corollary}

\section{Application to Type Inference}

\begin{definition}[Subtyping Check with Validation]
When checking $u_1 \leq u_2$ where $u_2$ contains type variable $T$:
\begin{enumerate}
\item Compute $\diff(u_1, u_2)$ and grow $T$'s lower bound.
\item Expand $u_1, u_2$ using the substitution.
\item Validate via $\mathsf{isSub}(\mathit{expand}(u_1), \mathit{expand}(u_2))$.
\end{enumerate}
\end{definition}

\begin{theorem}[Inference Soundness]
If $\mathsf{isSub}$ correctly validates subtyping on expanded types, then 
type inference is sound regardless of whether $\diff$ satisfies (P2).
\end{theorem}

\begin{proof}
The $\diff$ result is used only to guide inference (e.g. by accumulating bounds or hints about $T$).
The acceptance/rejection decision is made solely by $\mathsf{isSub}$ on expanded types.
Therefore, an imprecise $\diff$ can at worst cause incompleteness (e.g. by failing to discover a useful bound, leading to a later rejection), but it cannot cause an unsound acceptance when $\mathsf{isSub}$ is correct.
\end{proof}

\section{Examples of Incompleteness}

We present two concrete examples of incompleteness in SKJS:
\begin{itemize}
\item a \emph{control-flow refinement} case that TypeScript accepts but SKJS currently fails to narrow (a missing implementation case), and
\item a \emph{$\diff$-based} case where $\diff$ is incomplete.
\end{itemize}

\subsection{Example 1: Missing Strict-Equality Refinement for Primitive Literals}

\textbf{TypeScript code:}
\begin{verbatim}
function f(x: 1 | 2): 2 {
  if (x === 1) { return 2; }
  return x;  // x should be narrowed to 2
}
\end{verbatim}

\textbf{Mathematical setup:}
Both \texttt{1} and \texttt{2} have the same runtime tag $4$.
SKJS represents primitive literal unions by storing a finite literal-set payload inside a single tag representative, so \texttt{1 | 2} is representable and the narrowed type \texttt{2} is expressible.
Unlike Example~2, this is not a representational limitation.

\textbf{Incompleteness (missing implementation case):}
The control-flow refinement for strict equality in SKJS currently handles only:
\begin{enumerate}
\item algebraic discriminant comparisons (e.g.\ \texttt{x.\_tag === "Foo"}),
\item \texttt{typeof} comparisons, and
\item comparisons of a local to \texttt{null} or \texttt{undefined}.
\end{enumerate}
It does \emph{not} implement refinement for comparisons of a local against a primitive literal such as \texttt{x === 1}.
Therefore, SKJS does not narrow \texttt{x} in either branch and rejects \texttt{return x;}, even though TypeScript narrows \texttt{x} to \texttt{2}.

\paragraph{Could it be implemented?}
Yes. Since \texttt{1|2} is stored as a finite literal-set at the numeric tag, the true branch could intersect the payload with \(\{1\}\) and the false branch could subtract \(\{1\}\), yielding \(\{2\}\).
This requires adding a strict-equality refinement case for primitive literals, not changing the \(\STypes\) normal form.

\subsection{Example 2: Subtracting from Top (\texttt{unknown})}

\paragraph{Note on \texttt{\{\}} syntax.}
In TypeScript (with \texttt{--strictNullChecks}), the type \texttt{\{\}} behaves as a ``non-nullish top'': it includes all values except \texttt{null} and \texttt{undefined}, including primitives such as numbers and strings.
In SKJS, the surface syntax \texttt{\{\}} denotes an \emph{object} type (runtime tag $9$) and does \emph{not} include primitives.
The example below uses TypeScript's interpretation of \texttt{\{\}}.

\medskip
\textbf{TypeScript code (requires \texttt{--strictNullChecks}):}
\begin{verbatim}
function f(x: unknown): {} {
  if (x !== null && x !== undefined) {
    return x;  // x should be narrowed to {}
  }
  return {};
}
\end{verbatim}

\textbf{Mathematical setup:}
The type \texttt{unknown} is represented as $\clause{\emptyset}{\top}$.
Let $c_1 = \clause{\emptyset}{\top}$ represent \texttt{unknown}, and $c_2 = \clause{\emptyset}{\{\texttt{undefined}, \texttt{null}\}}$ represent \texttt{null | undefined}.

\textbf{Incompleteness:}
By definition of $\diff_{\Clause}$:
\[
\diff_{\Clause}(c_1, c_2) = c_1 \quad \text{(since } S_1 = \top\text{)}
\]
So $\sem{\diff_{\Clause}(c_1, c_2)} = \sem{c_1} = \Val$ (all values).
But the true set difference is:
\[
\sem{c_1} \setminus \sem{c_2} = \Val \setminus (\sem{\texttt{null}} \cup \sem{\texttt{undefined}}) \neq \Val
\]
The result is incomplete: $\diff$ returns \texttt{unknown} instead of the narrowed non-nullish type.
The clause diff cannot ``subtract from top'' because $\top$ does not enumerate tags---there is no finite set from which to remove the subtracted alternatives.
The implementation in \texttt{Clause::diff} returns the left clause unchanged when $S_1 = \top$.

\paragraph{Could it be implemented?}
Yes, but it requires a representation extension.
To model TypeScript's non-nullish type precisely, one could add a dedicated form (e.g.\ a \emph{non-nullish} flag, or an \emph{excluded-tag} mechanism) rather than attempting to enumerate ``all other tags'' inside \(\STypes\).
The current SKJS implementation chooses neither approach and leaves top unchanged when subtracting a non-top right-hand side.

\paragraph{Making this example work in SKJS.}
To express this pattern in SKJS, one would need a surface spelling for non-nullish types (such as \texttt{NonNullable<T>}) and an internal representation that can express ``remove null/undefined'' from an arbitrary type, including top.

\section{Implementation Correspondences}
\label{app:impl}

This appendix summarizes the correspondence between the mathematical model and SKJS implementation.

\begin{keydef}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}l@{\quad}l@{}}
\toprule
\textbf{Mathematical Concept} & \textbf{SKJS Implementation} \\
\midrule
$\SType$ (simple type) & \texttt{SType} enum with tag-indexed variants \\
$\STypes$ (tag-distinct set) & \texttt{STypes} struct, one entry per tag \\
$\clause{V}{S}$ (clause) & \texttt{Clause} record with \texttt{vars} and \texttt{stypes} \\
$\Clauses$ (clauses map) & \texttt{Clauses} keyed by variable-set \\
$\TType = (I, C)$ & \texttt{TType} with \texttt{insts} and \texttt{clauses} \\
$\refineop^{f,g}$ (filter-map) & \texttt{FilterType} transformer \\
$\diff$ (type difference) & \texttt{TType::diff}, \texttt{Clause::diff}, etc. \\
$\mathsf{single}(s)$ & Single-clause \texttt{TType} with one $\SType$ \\
$\mathsf{def}(R, d)$ (defaulting) & \texttt{setLocal} with \texttt{getSTypes} check \\
\bottomrule
\end{tabular}
\end{keydef}

\paragraph{Key implementation behaviors modeled.}
\begin{enumerate}
\item \textbf{Tag-indexed storage}: $\STypes$ stores at most one $\SType$ per runtime tag.
\item \textbf{Same-tag merge}: Union/intersection on $\STypes$ merge same-tag alternatives (Definition~\ref{def:stypes-union-inter}).
\item \textbf{Tag-elimination diff}: $\diff_{\STypes}$ removes whole tags, not same-tag subsets.
\item \textbf{Top-clause behavior}: Refinement and diff on $\clause{\emptyset}{\top}$ behave specially.
\item \textbf{Call-signature modes}: $\mathsf{None}$/$\mathsf{Maybe}$/$\mathsf{Definite}$ for callable objects.
\end{enumerate}
